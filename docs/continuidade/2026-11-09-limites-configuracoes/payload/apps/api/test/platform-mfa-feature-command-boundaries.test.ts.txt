function signal() { let resolve!: () => void; const promise = new Promise<void>(done => { resolve = done; }); return { promise, resolve }; }
async function bounded<T>(promise: PromiseLike<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([Promise.resolve(promise), new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`Fixture deadline: ${label}`)), 5000);
  })]); } finally { clearTimeout(timer); }
}
async function settleAll(transactions: Promise<unknown>[]) {
  await bounded((async () => {
    let completed = 0;
    while (completed < transactions.length) {
      const batch = transactions.slice(completed); completed += batch.length; await Promise.allSettled(batch);
    }
  })(), "all actual app transactions settled");
}
function trackAllTransactions() {
  const original = appDb.transaction, run = original.bind(appDb), transactions: Promise<unknown>[] = [];
  appDb.transaction = (callback, config) => { const pending = run(callback, config); transactions.push(pending); return pending; };
  return { settle: () => settleAll(transactions), restore() { appDb.transaction = original; } };
}

function observeBoundary(a:Actor,stage:Boundary,building:string) {

  let matched=false,capabilitySeen=false;
  type Input=Parameters<AppTransaction["execute"]>[0];
  appDb.transaction=(callback,config)=>{
    const pending=run(tx=>callback(new Proxy(tx,{get(target,key){
      if(key==="execute")return async(query:Input)=>{
        const compiled=typeof query==="string"?{sql:query,params:[]}:dialect.sqlToQuery(query.getSQL());
        const rows=await target.execute(query);
        const capability=compiled.sql.trim()==="select app_has_global_capability($1) as allowed"&&compiled.params.length===1&&compiled.params[0]==="features:manage";
        if(capability){assert.equal((rows[0] as {allowed?:unknown}|undefined)?.allowed,true);capabilitySeen=true;}
        const at=stage==="context"?compiled.sql.includes("set_config('app.user_id'")&&compiled.sql.includes("set_config('app.session_id'")&&compiled.params[0]===a.id&&compiled.params[2]===a.sid
          :stage==="capability"?capability:compiled.sql.trim()==="select app_can_discover_building($1) as allowed"&&compiled.params.length===1&&compiled.params[0]===building;
        if(!matched&&at){if(stage==="discovery")assert.equal((rows[0] as {allowed?:unknown}|undefined)?.allowed,true);matched=true;reached.resolve();await bounded(release.promise,"actual SQL boundary release");}
        return rows;
      };
      const value=Reflect.get(target,key,target);return typeof value==="function"?value.bind(target):value;
    }})),config);transactions.push(pending);return pending;
  };
  return {reached,release,settle:()=>settleAll(transactions),capability:()=>capabilitySeen,restore(){appDb.transaction=original;}};
}
