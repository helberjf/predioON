import { Router, type Request } from "express";
import { sql } from "drizzle-orm";
import { z } from "zod";
import { lockFeatureChanges, type AppTransaction } from "@predioon/db/runtime";
import { FEATURE_CATALOG, FEATURE_KEYS, FeatureKeySchema, FeatureUpdateSchema, REALTIME_CHANNEL, resolveFeatures, type FeatureKey, type FeatureStates } from "@predioon/shared";
import { assertBuildingDiscovery, assertGlobalCapability, currentAuth, inTenantContext } from "../../auth/middleware.js";
import { decodeMfaReadDate, readMfaRows } from "../../auth/mfa-read.js";
import { badRequest, conflict, forbidden, notFound } from "../../http/errors.js";
import { param } from "../../http/params.js";
import { validateBody } from "../../http/validate.js";
import { recordAudit } from "../audit/repo.js";

export const featuresRouter = Router();
featuresRouter.use((_req, res, next) => { res.setHeader("Cache-Control", "no-store"); next(); });

// PostgreSQL JSON bypasses Drizzle column decoding. These strict transport
// decoders preserve the public read and command response contracts.
const GlobalFeatureReadRow = z.object({
  key: FeatureKeySchema, enabled: z.boolean(), version: z.number().int(),
}).strict();
const LocalFeatureReadRow = GlobalFeatureReadRow.extend({ enabled: z.boolean().nullable() }).strict();
const FeatureRuntimeReadRow = z.object({
  key: FeatureKeySchema,
  resumedAt: z.string().nullable().transform(value => value === null ? null : decodeMfaReadDate(value).toISOString()),
}).strict();
const BuildingFeaturesReadRow = z.object({
  exists: z.boolean(), global: z.array(GlobalFeatureReadRow), local: z.array(LocalFeatureReadRow),
  runtime: z.array(FeatureRuntimeReadRow),
}).strict();

async function assertAdmin(tx: AppTransaction): Promise<void> {
  await assertGlobalCapability(tx, "features:manage");
}
async function assertBuilding(tx: AppTransaction, buildingId: string): Promise<void> {
  await assertBuildingDiscovery(tx, buildingId);
  const [building] = await readMfaRows(tx, sql`select id from public.buildings where id=${buildingId}`,
    value => z.object({ id: z.string() }).strict().parse(value), ["id"]);
  if (!building) throw notFound("Condomínio não encontrado");
}
async function globalView(tx: AppTransaction) {
  const rows = await readMfaRows(tx, sql`select feature_key as "key", enabled, version from public.global_feature_settings`,
    value => GlobalFeatureReadRow.parse(value), ["key"]);
  const states = resolveFeatures(rows, [], "global");
  return { items: Object.values(states).map(state => ({ ...state, version: state.globalVersion })) };
}
function featureParam(req: Request): FeatureKey {
  const parsed = FeatureKeySchema.safeParse(param(req, "key"));
  if (!parsed.success) throw badRequest("Funcionalidade inválida");
  return parsed.data;
}

featuresRouter.get("/catalog", async (req, res) => {
  await inTenantContext(req, async tx => {
    await assertAdmin(tx);
    // Revalidate the capability in the same snapshot as MFA before returning
    // the static catalog. A later grant change belongs to a later snapshot.
    const [row] = await tx.execute<{ allowed: unknown }>(sql`
      select case when public.app_require_session_mfa()
        then public.app_has_global_capability(${"features:manage"}) else null end as allowed
    `);
    if (!row || typeof row.allowed !== "boolean") throw new Error("Invalid guarded feature authorization");
    if (!row.allowed) throw forbidden("Sem a capacidade global necessária");
  });
  res.json({ items: FEATURE_CATALOG });
});
featuresRouter.get("/global", async (req, res) => {
  res.json(await inTenantContext(req, async tx => {
    await assertAdmin(tx);
    const global = await readMfaRows(tx,
      sql`select feature_key as "key", enabled, version from public.global_feature_settings`,
      value => GlobalFeatureReadRow.parse(value), ["key"],
    );
    const states = resolveFeatures(global, [], "global");
    return { items: Object.values(states).map(state => ({ ...state, version: state.globalVersion })) };
  }));
});
featuresRouter.get("/buildings/:buildingId", async (req, res) => {
  const buildingId = param(req, "buildingId");
  res.json(await inTenantContext(req, async tx => {
    await assertBuildingDiscovery(tx, buildingId);
    return buildingView(tx, buildingId);
  }));
});

async function buildingView(tx: AppTransaction, buildingId: string) {
  // A scalar envelope keeps existence, settings and runtime on the guarded
  // statement's snapshot, even for a missing building or empty settings.
  // Existing shared/exclusive feature locks still coordinate HTTP writes.
  const rows = await readMfaRows(tx, sql`
    select exists(select 1 from public.buildings where id = ${buildingId}) as "exists",
      coalesce((select jsonb_agg(jsonb_build_object(
        'key', g.feature_key, 'enabled', g.enabled, 'version', g.version
      ) order by g.feature_key) from public.global_feature_settings g), '[]'::jsonb) as "global",
      coalesce((select jsonb_agg(jsonb_build_object(
        'key', l.feature_key, 'enabled', l.enabled, 'version', l.version
      ) order by l.feature_key) from public.building_feature_settings l
        where l.building_id = ${buildingId}), '[]'::jsonb) as "local",
      coalesce((select jsonb_agg(jsonb_build_object(
        'key', r.feature_key, 'resumedAt', r.resumed_at
      ) order by r.feature_key) from public.feature_runtime r
        where r.building_id = ${buildingId}), '[]'::jsonb) as "runtime"
  `, value => BuildingFeaturesReadRow.parse(value), ["exists"]);
  const row = rows[0];
  if (rows.length !== 1 || !row) throw new Error("Invalid guarded feature result");
  if (!row.exists) throw notFound("Condomínio não encontrado");
  const states = resolveFeatures(row.global, row.local);
  for (const runtime of row.runtime) if (states[runtime.key]) states[runtime.key].resumedAt = runtime.resumedAt;
  return { items: Object.values(states) };
}

async function assertFeatureCommand(tx: AppTransaction): Promise<void> {
  const [row] = await tx.execute<{ allowed: unknown }>(sql`
    select case when public.app_require_session_mfa()
      then public.app_has_global_capability(${"features:manage"}) else null end as allowed
  `);
  if (!row || typeof row.allowed !== "boolean") throw new Error("Invalid guarded feature authorization");
  if (!row.allowed) throw forbidden("Sem a capacidade global necessária");
}

async function commandStates(tx: AppTransaction, buildingId: string): Promise<FeatureStates> {
  const view = await buildingView(tx, buildingId);
  return Object.fromEntries(view.items.map(state => [state.key, state])) as FeatureStates;
}

async function update(req: Request, buildingId: string | null) {
  const key = featureParam(req), input = FeatureUpdateSchema.parse(req.body);
  if (buildingId === null && input.enabled === null) throw badRequest("A configuração global deve ser ligada ou desligada");
  return inTenantContext(req, async tx => {
    await lockFeatureChanges(tx);
    await assertAdmin(tx);
    await assertFeatureCommand(tx);
    if (buildingId !== null) await assertBuilding(tx, buildingId);
    const targets = buildingId === null ? await readMfaRows(tx, sql`select id from public.buildings`,
      value => z.object({ id: z.string() }).strict().parse(value), ["id"]) : [{ id: buildingId }];
    const priorStates = new Map<string, FeatureStates>();
    for (const target of targets) priorStates.set(target.id, await commandStates(tx, target.id));
    const [prior] = buildingId === null
      ? await readMfaRows(tx, sql`select feature_key as "key", enabled, version from public.global_feature_settings where feature_key=${key}`,
        value => GlobalFeatureReadRow.parse(value), ["key"])
      : await readMfaRows(tx, sql`select feature_key as "key", enabled, version from public.building_feature_settings where building_id=${buildingId} and feature_key=${key}`,
        value => LocalFeatureReadRow.parse(value), ["key"]);
    if ((prior?.version ?? 0) !== input.version) throw conflict("Configuração alterada por outro administrador. Atualize a tela.", { code: "FEATURE_VERSION_CONFLICT" });
    const version = input.version + 1;
    // Guard the upsert itself, including an empty INSERT SELECT. The existing
    // exclusive advisory lock still serializes cooperating configuration writes.
    const mutation = buildingId === null ? sql`
      insert into public.global_feature_settings(feature_key,enabled,version,updated_at)
      select ${key},${input.enabled},${version},clock_timestamp() from feature_mfa where allowed
      on conflict(feature_key) do update set enabled=excluded.enabled,version=excluded.version,updated_at=clock_timestamp()
      returning feature_key as "key",enabled,version
    ` : sql`
      insert into public.building_feature_settings(building_id,feature_key,enabled,version,updated_at)
      select ${buildingId},${key},${input.enabled},${version},clock_timestamp() from feature_mfa where allowed
      on conflict(building_id,feature_key) do update set enabled=excluded.enabled,version=excluded.version,updated_at=clock_timestamp()
      returning feature_key as "key",enabled,version
    `;
    const [result] = await tx.execute<{ allowed: unknown; rows: unknown }>(sql`
      with feature_mfa as materialized (
        select case when public.app_require_session_mfa()
          then public.app_has_global_capability(${"features:manage"}) else null end as allowed
      ), written as (${mutation})
      select allowed,case when allowed then (select coalesce(jsonb_agg(to_jsonb(written)), '[]'::jsonb) from written)
        else null end as rows from feature_mfa
    `);
    // ON CONFLICT can wait on a row changed outside the advisory-lock protocol.
    // A fresh statement must validate identity before decoding any outcome.
    await assertFeatureCommand(tx);
    if (!result || typeof result.allowed !== "boolean") throw new Error("Invalid guarded feature mutation");
    if (!result.allowed) throw forbidden("Sem a capacidade global necessária");
    if (!Array.isArray(result.rows) || result.rows.length !== 1) throw new Error("Feature configuration was not persisted");
    const written = (buildingId === null ? GlobalFeatureReadRow : LocalFeatureReadRow).parse(result.rows[0]);
    if (written.key !== key || written.enabled !== input.enabled || written.version !== version) throw new Error("Invalid persisted feature configuration");
    let affected = 0;
    for (const target of targets) {
      const after = await commandStates(tx, target.id), before = priorStates.get(target.id)!;
      let changed = false;
      for (const feature of FEATURE_KEYS) if (before[feature].enabled !== after[feature].enabled) {
        // This VOLATILE helper has internal writes and snapshots. Guard entry,
        // then revalidate on return so every effect rolls back with the command.
        await tx.execute(sql`select case when public.app_require_session_mfa()
          then public.app_apply_feature_transition(${target.id}, ${feature}, ${after[feature].enabled}) else null end`);
        await assertFeatureCommand(tx);
        changed = true;
      }
      if (changed) affected++;
    }
    await recordAudit(tx, req, { buildingId, userId: currentAuth(req).userId, action: "FEATURE_CONFIGURATION_CHANGED", resourceType: "feature", resourceId: key,
      metadata: { scope: buildingId === null ? "GLOBAL" : "BUILDING", before: prior?.enabled ?? (buildingId === null ? true : null), after: input.enabled, version, reason: input.reason, affectedBuildings: affected } });
    await assertFeatureCommand(tx);
    // NOTIFY is transactional: a failed final check also discards its delivery.
    await tx.execute(sql`select case when public.app_require_session_mfa()
      then pg_notify(${REALTIME_CHANNEL}, ${JSON.stringify({ kind: "features-changed", buildingId: buildingId ?? "*" })}) else null end`);
    const view = buildingId === null ? await globalView(tx) : await buildingView(tx, buildingId);
    await assertFeatureCommand(tx);
    return view;
  }, { featureWrite: true });
}

featuresRouter.put("/global/:key", validateBody(FeatureUpdateSchema), async (req, res) => res.json(await update(req, null)));
featuresRouter.put("/buildings/:buildingId/:key", validateBody(FeatureUpdateSchema), async (req, res) => res.json(await update(req, param(req, "buildingId"))));
