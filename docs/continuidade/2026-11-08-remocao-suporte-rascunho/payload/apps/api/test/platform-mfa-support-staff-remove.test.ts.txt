function signal() { let resolve!: () => void; const promise = new Promise<void>(done => { resolve = done; }); return { promise, resolve }; }
async function bounded<T>(promise: PromiseLike<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([Promise.resolve(promise), new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`Fixture deadline: ${label}`)), 5000);
  })]); } finally { clearTimeout(timer); }
}
async function settleAll(transactions: Promise<unknown>[]) {
  await bounded((async () => {
    let completed = 0;
    while (completed < transactions.length) {
      const batch = transactions.slice(completed); completed += batch.length; await Promise.allSettled(batch);
    }
  })(), "all actual app transactions settled");
}
function trackAllTransactions() {
  const original = appDb.transaction, run = original.bind(appDb), transactions: Promise<unknown>[] = [];
  appDb.transaction = (callback, config) => { const pending = run(callback, config); transactions.push(pending); return pending; };
  return { settle: () => settleAll(transactions), restore() { appDb.transaction = original; } };
}
function observeAuthorization() {
  const original = appDb.transaction, run = original.bind(appDb), dialect = new PgDialect();
  const reached = signal(), release = signal(), transactions: Promise<unknown>[] = [];
  let matched = false;
  type Input = Parameters<AppTransaction["execute"]>[0];
  appDb.transaction = (callback, config) => {
    const pending = run(tx => callback(new Proxy(tx, { get(target, key) {
      if (key === "execute") return async (query: Input) => {
        const compiled = typeof query === "string" ? { sql: query, params: [] } : dialect.sqlToQuery(query.getSQL());
        const rows = await target.execute(query);
        // Pause AFTER the actual original global capability returned true.
        if (!matched && compiled.sql.trim() === "select app_has_global_capability($1) as allowed" && compiled.params.length === 1 && compiled.params[0] === "rbac:manage") {
          assert.equal((rows[0] as { allowed?: unknown } | undefined)?.allowed, true);
          matched = true; reached.resolve(); await bounded(release.promise, "real authorization release");
        }
        return rows;
      };
      const value = Reflect.get(target, key, target); return typeof value === "function" ? value.bind(target) : value;
    }})), config);
    transactions.push(pending); return pending;
  };
  return { reached, release, settle: () => settleAll(transactions), restore() { appDb.transaction = original; } };
}

    await sqlClient`insert into role_bindings(id,user_id,building_id,role_key) values(${wrong},${a.id},${building},'BUILDING_ADMIN')`;
    for(const id of ["not-a-uuid",randomUUID(),wrong])assert.equal((await request(a,id)).status,404);
    assert.equal((await http(undefined,"/v1/support-access/staff/"+op.id,"DELETE")).status,401);assert.deepEqual(await snapshot(a,op),before);
  });
  it("repeated removal creates no additional staff or grant audit",async()=>{
    const a=await actor("flag"),op=await operation("active");await effects(a,op,await request(a,op.id));const before=await snapshot(a,op);
    assert.equal((await request(a,op.id)).status,204);assert.deepEqual(await snapshot(a,op),before);
  });
  it("removal does not require recent password verification",async()=>{
    const a=await actor("binding"),op=await operation("active");await sqlClient`update sessions set verified_at=clock_timestamp()-interval '11 minutes' where id=${a.sid}::uuid`;
    await effects(a,op,await request(a,op.id));
  });
  for(const relation of ["role_bindings","support_grants","audit_logs"] as const)for(const mode of ["proof","exception"] as const){
    it(`${relation}: rolls staff and all grant removals back after ${mode} trigger`,async t=>{
      const a=await actor("flag"),control=await operation("active");await effects(a,control,await request(a,control.id));
      const op=await operation("active"),before=await snapshot(a,op),originalProof=await proof(a),catalog=await triggerCatalog(),name="tstaffremove_"+randomUUID().replaceAll("-","");
      assert.equal(originalProof.verified,true);assert.equal(typeof originalProof.verified_at,"string");assert.equal(typeof originalProof.generation,"string");
      try{await installTrigger(name,relation,a,op,mode);const response=await request(a,op.id);await witness(name);
        if(mode==="proof"){
          if(response.status===204){await effects(a,op,response);assert.deepEqual(await proof(a),{verified:false,verified_at:null,generation:null});
            t.diagnostic("Baseline204: staff and grant mutations and audits persisted; narrow trigger fired once and removed the same family proof");}
          else{assert.deepEqual(await snapshot(a,op),before);assert.deepEqual(await proof(a),originalProof);}
          denied(response,"MFA_VERIFICATION_REQUIRED");
        }else{assert.equal(response.status,500);assert.equal(JSON.stringify(response.body).includes("private staff removal fixture detail"),false);}
        assert.deepEqual(await snapshot(a,op),before);assert.deepEqual(await proof(a),originalProof);
      }finally{await tracking?.settle();await dropTrigger(name);assert.deepEqual(await triggerCatalog(),catalog);}
    });
  }

    }finally{await tracking?.settle();await dropTrigger(name);assert.deepEqual(await triggerCatalog(),catalog);}
  });
