// SAME SOURCE baseline/fixed: original authorization and SQL execute normally.
// These tests arrange synthetic MFA proof after real HTTP login. They do not
// replace the existing real-TOTP or SQL-ACL regression tests.
type Kind = "blocks" | "units" | "teams";
type Target = { id: string; kind: Kind; buildingId: string };
type Actor = { id: string; token: string; sid: string; local: string; global: string; generation: string };
type Row = Record<string, unknown>;
type Response = { status: number; body: unknown };
type Command = { target: Target; status: boolean; body: Row };
type Boundary = "capability" | "context";
const kinds: Kind[] = ["blocks", "units", "teams"];
const capabilities = { blocks: "units:manage", units: "units:manage", teams: "teams:manage" };
const prefixes = { blocks: "BLOCK", units: "UNIT", teams: "TEAM" };
const resourceTypes = { blocks: "block", units: "unit", teams: "team" };
function signal() { let resolve!: () => void; const promise = new Promise<void>(done => { resolve = done; }); return { promise, resolve }; }
async function bounded<T>(promise: PromiseLike<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([Promise.resolve(promise), new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`Fixture deadline: ${label}`)), 5000);
  })]); } finally { clearTimeout(timer); }
}

async function settleAll(transactions: Promise<unknown>[]) {
  await bounded((async () => {
    let completed = 0;
    while (completed < transactions.length) {
      const batch = transactions.slice(completed); completed += batch.length;
      await Promise.allSettled(batch);
    }
  })(), "all actual app transactions settled");
}
function trackAllTransactions() {
  const original = appDb.transaction, run = original.bind(appDb), transactions: Promise<unknown>[] = [];
  appDb.transaction = (callback, config) => { const pending = run(callback, config); transactions.push(pending); return pending; };
  return { settle: () => settleAll(transactions), restore() { appDb.transaction = original; } };
}

function observe(command: Command, boundary: Boundary, pause = true) {
  const original = appDb.transaction, run = original.bind(appDb), dialect = new PgDialect();
  const reached = signal(), release = signal(), transactions: Promise<unknown>[] = [];
  let matched = false, pid: number | undefined;
  type Input = Parameters<AppTransaction["execute"]>[0];
  appDb.transaction = (callback, config) => {
    const transaction = run(tx => callback(new Proxy(tx, { get(target, key) {
      if (key === "execute") return async (query: Input) => {
        const compiled = typeof query === "string" ? { sql: query, params: [] } : dialect.sqlToQuery(query.getSQL());
        const result = await target.execute(query);
        const match = boundary === "context" ? compiled.sql.includes("pg_advisory_xact_lock_shared(814772,")
          : compiled.sql.includes("app_has_capability(") && compiled.params.includes(capabilities[command.target.kind]) && compiled.params.includes(command.target.buildingId);
        if (!matched && match) {
          matched = true;
          const [connection] = await target.execute<{ pid: number }>(sql`select pg_backend_pid() as pid`);
          assert.ok(connection); pid = connection.pid; reached.resolve();
          if (pause) await bounded(release.promise, "authorization boundary release");
        }
        return result;
      };
      const value = Reflect.get(target, key, target); return typeof value === "function" ? value.bind(target) : value;
    }})), config);
    transactions.push(transaction); return transaction;
  };
  async function settleTransactions() {
    // SQL refusal/rollback is an expected HTTP result, not a second error.
    await settleAll(transactions);
  }
  return { reached, release, settleTransactions, get pid() { return pid; }, restore() { appDb.transaction = original; } };
}

  async function interleave(operation: Command, boundary: Boundary, change: () => Promise<unknown>) {
    const observation = observe(operation, boundary), controller = new AbortController();

      await bounded(change(), "identity change committed"); observation.release.resolve(); return await pending;
    } catch (error) { failed = true; throw error; }
    finally {
      observation.release.resolve(); controller.abort();
      const results = await Promise.allSettled([bounded(pending, "HTTP settled"), observation.settleTransactions()]);
      observation.restore();
      if (!failed) { const failure = results.find(result => result.status === "rejected"); if (failure?.status === "rejected") throw failure.reason; }
    }
  }
  async function whileRowLocked(operation: Command, change: () => Promise<unknown>) {
    const locked = signal(), release = signal(); let holderPid: number | undefined;
    const holder = sqlClient.begin(async owner => {
      await owner`set local lock_timeout='2s'`; await owner`set local statement_timeout='3s'`;
      const [pid] = await owner`select pg_backend_pid() as pid`; holderPid = pid!.pid;
      await owner`select id from ${sqlClient(operation.target.kind)} where id=${operation.target.id}::uuid for update`;
      locked.resolve(); await bounded(release.promise, "owner row lock release");
    });
    void holder.catch(() => undefined);
    const observation = observe(operation, "capability", false), controller = new AbortController();
    let pending: Promise<Response> | undefined, failed = false;
    try {
      assert.equal(await bounded(Promise.race([locked.promise.then(() => "locked"), Promise.resolve(holder).then(() => "holder ended")]), "owner lock acquired"), "locked");
      pending = request(admin, operation, controller); void pending.catch(() => undefined);
      assert.equal(await bounded(Promise.race([observation.reached.promise.then(() => "capability"), pending.then(() => "response")]), "real capability before lock"), "capability");
      const appPid = observation.pid; assert.equal(typeof appPid, "number"); assert.equal(typeof holderPid, "number");
      await bounded((async () => {
        const deadline = Date.now() + 4500;
        while (Date.now() < deadline) {
          const [waiting] = await sqlClient`select ${holderPid!}::int=any(pg_blocking_pids(${appPid!}::int)) as waiting`;
          if (waiting?.waiting) return;
          await new Promise<void>(resolve => setTimeout(resolve, 10));
        }
        throw new Error("Real app connection did not wait on the owner row lock");
      })(), "PostgreSQL row lock waiter");
      await bounded(change(), "identity change during real row wait");
      release.resolve(); await bounded(holder, "owner transaction released"); return await pending;
    } catch (error) { failed = true; throw error; }
    finally {
      release.resolve(); observation.release.resolve(); controller.abort();
      const results = await Promise.allSettled([bounded(holder, "owner holder settled"), ...(pending ? [bounded(pending, "HTTP settled")] : []), observation.settleTransactions()]);
      observation.restore();
      if (!failed) { const failure = results.find(result => result.status === "rejected"); if (failure?.status === "rejected") throw failure.reason; }
    }
  }

        await sqlClient.unsafe(`create function public.${name}() returns trigger language plpgsql security definer
          set search_path=pg_catalog,pg_temp as $$ begin
          if new.resource_id=TG_ARGV[0] and new.user_id=TG_ARGV[1] and new.action=TG_ARGV[2]
             and current_setting('app.session_id',true)=TG_ARGV[3] then
            perform pg_catalog.nextval('public.${name}_seq'::regclass);
            update public.sessions set mfa_verified_at=null,mfa_factor_generation=null
              where id=TG_ARGV[3]::uuid and user_id=TG_ARGV[1];
          end if; return new; end $$`);
        await sqlClient.unsafe(`revoke all on function public.${name}() from public`);
        // The arguments are server-generated IDs/action, never user-controlled SQL.
        const args = [operation.target.id, admin.id, action(operation), admin.sid];
        assert.ok(args.every(value => /^[A-Za-z0-9_-]+$/.test(value)));
        await sqlClient.unsafe(`create trigger ${name} after insert on public.audit_logs for each row execute function public.${name}(${args.map(value => "'" + value + "'").join(",")})`);
        const [metadata] = await sqlClient`select p.prosecdef,p.proconfig,pg_get_userbyid(p.proowner)=current_user as owner,
          not exists(select 1 from aclexplode(p.proacl) a where a.grantee=0 and a.privilege_type='EXECUTE') as public_revoked
          from pg_proc p join pg_namespace n on n.oid=p.pronamespace where n.nspname='public' and p.proname=${name}`;
        assert.equal(metadata?.prosecdef, true); assert.equal(metadata?.owner, true); assert.equal(metadata?.public_revoked, true);
        assert.deepEqual(metadata?.proconfig, ["search_path=pg_catalog, pg_temp"]);
        const response = await request(admin, operation);
        const [counter] = await sqlClient.unsafe(`select last_value::int,is_called from public.${name}_seq`);
        assert.equal(counter?.is_called, true); assert.equal(counter?.last_value, 1, "exactly one audit-trigger execution, survives transaction rollback");
        child.diagnostic("Narrow audit fixture trigger executed exactly once for actor/resource/action/SID; sequence proves execution even on rollback");
        await classify(admin, operation, original, response, text => child.diagnostic(text));
        const [session] = await sqlClient`select mfa_verified_at is not null as verified,mfa_factor_generation::text as generation from sessions where id=${admin.sid}`;
        if (response.status === 200) { assert.equal(session?.verified, false); assert.equal(session?.generation, null); }
        else { assert.equal(session?.verified, true); assert.equal(session?.generation, admin.generation, "trigger session write rolled back with mutation/audit"); }
        denied(response, "MFA_VERIFICATION_REQUIRED"); await unchanged(operation.target, original);
      } finally {
        try { await tracking?.settle(); await dropTrigger(name); assert.deepEqual(await triggerCatalog(), initialCatalog); }
        finally { await bounded(prove(admin), "restore audit-case session"); }
      }
    });
  });
