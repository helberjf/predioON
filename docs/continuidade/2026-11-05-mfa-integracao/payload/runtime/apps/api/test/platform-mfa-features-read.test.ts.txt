// Identical test source on baseline/fixed; no import of the new read helper.
// Owner mutations only arrange fixture authority/settings. The serialization
// control below uses a real HTTP PUT, which respects the feature advisory lock.
type Actor = { id: string; token: string; sid: string; binding: string; generation: string };
type HttpResult = { status: number; body: unknown; cache: string | null };
type Compiled = { sql: string; params: unknown[] };
type Boundary = { functionName: string; parameter: string };
type GlobalRow = { feature_key: string; enabled: boolean; version: number; updated_at: string };

function signal() {
  let resolve!: () => void;
  const promise = new Promise<void>(done => { resolve = done; });
  return { promise, resolve };
}
async function bounded<T>(promise: PromiseLike<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try {
    return await Promise.race([Promise.resolve(promise), new Promise<never>((_, reject) => {
      timer = setTimeout(() => reject(new Error(`Fixture deadline: ${label}`)), 5000);
    })]);
  } finally { clearTimeout(timer); }
}

function observeTransactions(boundary?: Boundary, observeExclusive = false) {
  const original = appDb.transaction, run = original.bind(appDb), dialect = new PgDialect();
  const reached = signal(), release = signal(), exclusiveAttempt = signal();
  const delegated: Compiled[] = [];
  const transactions: Promise<unknown>[] = [];
  let contexts = 0, exclusivePid: number | undefined, exclusiveAcquired = false;
  type Input = Parameters<AppTransaction["execute"]>[0];
  appDb.transaction = (callback, config) => {
    const transaction = run(tx => callback(new Proxy(tx, { get(target, key) {
    if (key === "execute") return async (query: Input) => {
      const compiled: Compiled = typeof query === "string" ? { sql: query, params: [] } : dialect.sqlToQuery(query.getSQL());
      const exclusive = observeExclusive && /pg_advisory_xact_lock\(814772,\s*1\)/.test(compiled.sql);
      if (exclusive) {
        const [pid] = await target.execute<{ pid: number }>(sql`select pg_backend_pid() as pid`);
        assert.ok(pid); exclusivePid = pid.pid; exclusiveAttempt.resolve();
      }
      // Always execute the original operation, including real authorization,
      // before announcing the boundary. Never synthesize an allowed row.
      const rows = await target.execute(query);
      delegated.push(compiled);
      if (exclusive) exclusiveAcquired = true;
      if (boundary && contexts === 0 && compiled.sql.includes(boundary.functionName + "(") && compiled.params.includes(boundary.parameter)) {
        contexts++; reached.resolve(); await bounded(release.promise, "projection release");
      }
      return rows;
    };
    const value = Reflect.get(target, key, target);
    return typeof value === "function" ? value.bind(target) : value;
    }})), config);
    transactions.push(transaction);
    return transaction;
  };
  async function settleTransactions(): Promise<void> {
    await bounded((async () => {
      let completed = 0;
      // Track every actual transaction, including a writer still running after
      // its HTTP fetch is aborted. New transactions registered while waiting
      // are drained in the next batch before the observer can be restored.
      while (completed < transactions.length) {
        const batch = transactions.slice(completed); completed += batch.length;
        // SQL MFA refusals are expected in negative HTTP scenarios. Wait for
        // their rollback without replacing the response/status/reason oracle.
        await Promise.allSettled(batch);
      }
    })(), "all real transactions settled");
  }
  return {
    reached, release, exclusiveAttempt, delegated, settleTransactions,
    get contexts() { return contexts; }, get exclusivePid() { return exclusivePid; },
    get exclusiveAcquired() { return exclusiveAcquired; },
    restore() { appDb.transaction = original; },
  };
}

describe("MFA feature catalog, global and building reads", { concurrency: false }, () => {

  async function request(a: Actor, path: string, controller = new AbortController(), method = "GET", body?: unknown): Promise<HttpResult> {
    try {
      const response = await bounded(fetch(url + path, {
        method, headers: { Authorization: `Bearer ${a.token}`, "Content-Type": "application/json" },
        signal: controller.signal, ...(body === undefined ? {} : { body: JSON.stringify(body) }),
      }), "HTTP");
      return { status: response.status, body: await bounded(response.json(), "JSON"), cache: response.headers.get("cache-control") };
    } finally { controller.abort(); }
  }
  async function actor(label: string, local: boolean): Promise<Actor> {
    const id = `mfr-${suffix}-${label}`, binding = randomUUID(), generation = randomUUID(); users.push(id);
    await sqlClient`insert into users(id,name,email,password_hash) values(${id},${label},${id + "@example.invalid"},${hash})`;

      await sqlClient`insert into role_bindings(user_id,building_id,role_key) values(${id},${idOfBuilding},'BUILDING_ADMIN')`;
    }
    await sqlClient`insert into role_bindings(id,user_id,role_key,active) values(${binding},${id},'PLATFORM_ADMIN',false)`;
    const controller = new AbortController();
    try {
      const response = await bounded(fetch(url + "/auth/login", {
        method: "POST", headers: { "Content-Type": "application/json" }, signal: controller.signal,
        body: JSON.stringify({ email: id + "@example.invalid", password }),
      }), "login");
      assert.equal(response.status, 200);
      const data = await bounded(response.json(), "login JSON") as { accessToken: string };
      const sid = decodeJwt(data.accessToken).sid; assert.equal(typeof sid, "string");
      return { id, token: data.accessToken, sid: sid as string, binding, generation };
    } finally { controller.abort(); }
  }
  async function prove(a: Actor) {
    await sqlClient`update sessions set mfa_verified_at=now(),mfa_factor_generation=${a.generation} where id=${a.sid}`;
  }
  async function restoreFactor(a: Actor) {
    await sqlClient`insert into mfa_factors(user_id,secret_ciphertext,confirmed_at,generation)
      values(${a.id},decode(repeat('00',40),'hex'),now(),${a.generation})
      on conflict(user_id) do update set confirmed_at=excluded.confirmed_at,generation=excluded.generation`;
    await prove(a);
  }
  async function satisfied(a: Actor) {
    const response = await request(a, "/auth/me");
    assert.equal(response.status, 200);
    assert.equal((response.body as { mfaRequirement: string }).mfaRequirement, "satisfied");
    assert.equal(decodeJwt(a.token).sid, a.sid, "same bearer and SID, no renewed login");
  }
  function states(response: HttpResult): FeatureState[] {
    assert.equal(response.status, 200, JSON.stringify(response.body));
    assert.equal(response.cache, "no-store");
    assert.deepEqual(Object.keys(response.body as object), ["items"]);
    const items = (response.body as { items: FeatureState[] }).items;
    assert.equal(items.length, 24);
    assert.deepEqual(items.map(row => row.key), FEATURE_CATALOG.map(row => row.key));
    for (const row of items) assert.deepEqual(Object.keys(row).sort(), fields);
    return items;
  }
  function defaults(scope: "building" | "global"): FeatureState[] {
    return FEATURE_CATALOG.map(({ key, optIn }) => ({
      key, enabled: !(scope === "building" && optIn), globalEnabled: true, localEnabled: null,
      blockedBy: scope === "building" && optIn ? "OPT_IN" : null,
      version: 0, globalVersion: 0, resumedAt: null,
    }));
  }
  function byKey(items: FeatureState[], key: FeatureKey) {
    const row = items.find(item => item.key === key); assert.ok(row); return row;
  }
  function denied(response: HttpResult, reason: string) {
    assert.equal(response.status, 403);
    assert.equal((response.body as { details?: { reason?: string } }).details?.reason, reason);
    assert.equal("items" in (response.body as object), false);
  }
  function classify(response: HttpResult, positive: HttpResult, diagnostic: (message: string) => void) {
    if (response.status === 200) {
      assert.deepEqual(response.body, positive.body);
      diagnostic("Baseline returned the exact decoded positive DTO after the authority change");
    } else if (response.status === 404) {
      assert.deepEqual(response.body, positive.body);
      diagnostic("Baseline returned the exact no-match404 body; no private DTO delivery claimed");
    } else diagnostic(`Observed HTTP${response.status}; the exact MFA reason is asserted separately`);
  }
  async function interleave(a: Actor, path: string, boundary: Boundary, change: () => Promise<unknown>) {
    const observation = observeTransactions(boundary), controller = new AbortController();
    const pending = request(a, path, controller); void pending.catch(() => undefined); let failed = false;
    try {
      assert.equal(await bounded(Promise.race([
        observation.reached.promise.then(() => "boundary"), pending.then(() => "response"),
      ]), "reach authorization boundary"), "boundary");
      await bounded(change(), "authority change committed");
      observation.release.resolve();
      return await pending;
    } catch (error) { failed = true; throw error; }
    finally {
      observation.release.resolve(); controller.abort();
      const results = await Promise.allSettled([bounded(pending, "HTTP settled")]);
      try { await observation.settleTransactions(); }
      finally { observation.restore(); }
      if (!failed) { const failure = results.find(result => result.status === "rejected"); if (failure?.status === "rejected") throw failure.reason; }
    }
  }
  async function observed(a: Actor, path: string, diagnostic: (message: string) => void) {
    const observation = observeTransactions();
    try {
      const response = await request(a, path);
      const guarded = observation.delegated.filter(query => query.sql.includes("app_require_session_mfa("));
      diagnostic(JSON.stringify({ path, delegatedGuardedExecuteCount: guarded.length, tablesInGuardedExecute: guarded.map(query => [
        "global_feature_settings", "building_feature_settings", "feature_runtime",
      ].filter(table => query.sql.includes(table))) }));
      // Observation only: baseline builders do not use this proxy's execute.
      // The semantic RED tests above remain identical and mandatory on both.
      return response;
    } finally {
      try { await observation.settleTransactions(); }
      finally { observation.restore(); }
    }
  }

      writing = request(admin, buildingPath(empty) + "/ENERGY_CONSUMPTION", writeController, "PUT", { enabled: false, version: 0, reason: "Prova de serialização HTTP" });
      void writing.catch(() => undefined);
      assert.equal(await bounded(Promise.race([observation.exclusiveAttempt.promise.then(() => "lock"), writing.then(() => "response")]), "HTTP writer reaches exclusive lock"), "lock");
      const pid = observation.exclusivePid; assert.equal(typeof pid, "number");
      // Observe PostgreSQL waiting, not an arbitrary sleep or guessed ordering.
      await bounded((async () => {
        const deadline = Date.now() + 4500;
        while (Date.now() < deadline) {
          const [waiting] = await sqlClient`select exists(select 1 from pg_locks
            where pid=${pid!} and locktype='advisory' and classid=814772 and objid=1 and objsubid=2
              and mode='ExclusiveLock' and not granted) as waiting`;
          if (waiting?.waiting) return;
          assert.equal(observation.exclusiveAcquired, false, "writer must not acquire while read is paused");
          await new Promise<void>(resolve => setTimeout(resolve, 10));
        }
        throw new Error("PostgreSQL did not expose the expected advisory lock waiter");
      })(), "PostgreSQL exclusive lock wait");
      assert.equal(observation.exclusiveAcquired, false);
      observation.release.resolve();
      assert.deepEqual((await reading).body, before.body, "first GET retains pre-write state");
      const written = states(await writing), next = states(await request(manager, buildingPath(empty)));
      assert.equal(observation.exclusiveAcquired, true);
      assert.equal(byKey(written, "ENERGY_CONSUMPTION").enabled, false);
      assert.equal(byKey(written, "ENERGY_CONSUMPTION").version, 1);
      assert.deepEqual(next, written);
      t.diagnostic("Real HTTP PUT waited on the read shared advisory lock; this is not an owner bypass or an intrastatement snapshot proof");
    } catch (error) { failed = true; throw error; }
    finally {
      observation.release.resolve(); readController.abort(); writeController.abort();
      const results = await Promise.allSettled([
        bounded(reading, "read settled"), ...(writing ? [bounded(writing, "write settled")] : []),
      ]);
      try { await observation.settleTransactions(); }
      finally { observation.restore(); }
      try {
        await bounded(sqlClient.begin(async tx => {
          await tx`set local lock_timeout='2s'`; await tx`set local statement_timeout='3s'`;
          await tx`delete from building_feature_settings where building_id=${empty} and feature_key='ENERGY_CONSUMPTION'`;
          await tx`delete from feature_runtime where building_id=${empty} and feature_key='ENERGY_CONSUMPTION'`;
        }), "serialization fixture cleanup");
      } finally {
        if (!failed) { const failure = results.find(result => result.status === "rejected"); if (failure?.status === "rejected") throw failure.reason; }
      }
    }
  });
});
