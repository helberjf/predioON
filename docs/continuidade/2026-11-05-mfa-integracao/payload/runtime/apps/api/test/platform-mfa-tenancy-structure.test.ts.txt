function signal() {
  let resolve!: () => void;
  const promise = new Promise<void>(done => { resolve = done; }); return { promise, resolve };
}
async function bounded<T>(promise: PromiseLike<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([Promise.resolve(promise), new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`Fixture deadline: ${label}`)), 5000);
  })]); } finally { clearTimeout(timer); }
}
function beforeProjection(capability:string) {
  const original = appDb.transaction, run = original.bind(appDb), dialect = new PgDialect();
  const reached = signal(), release = signal(), finished = signal(); let contexts = 0;
  type Input = Parameters<AppTransaction['execute']>[0];
  appDb.transaction = (callback, config) => run(tx => callback(new Proxy(tx, { get(target,key) {
    if (key === 'execute') return async (query: Input) => {
      const compiled = typeof query === 'string' ? {sql:query,params:[]} : dialect.sqlToQuery(query.getSQL());
      const text=compiled.sql;
      const rows = await target.execute(query);
      if (text.includes("app_has_capability(") && compiled.params.includes(capability)) {
        contexts++; assert.equal(contexts,1); reached.resolve(); await bounded(release.promise,'projection release');
      }
      return rows;
    };
    const value = Reflect.get(target,key,target); return typeof value === 'function' ? value.bind(target) : value;
  }})), config).finally(() => finished.resolve());
  return { reached,release,finished,get contexts(){return contexts;},restore(){appDb.transaction=original;} };
}

describe('MFA Batch A tenancy structure reads', {concurrency:false},()=>{
  const suffix=randomUUID(),org=`struct-org-${suffix}`,building=`struct-a-${suffix}`,foreign=`struct-b-${suffix}`;
  const routes=['blocks','units','teams'] as const;type Route=typeof routes[number];
  const users:string[]=[],password='structure-fictitious-password';let server:Server|undefined,url='',hash='';
  let manager:Actor,resident:Actor,satisfied:Actor;
  const ids:Record<Route,string[]>={blocks:[randomUUID(),randomUUID(),randomUUID()].sort(),units:[randomUUID(),randomUUID(),randomUUID()].sort(),teams:[randomUUID(),randomUUID(),randomUUID()].sort()};
  async function request(a:Actor,path:string,abort=new AbortController()):Promise<HttpResult>{
    try{const r=await bounded(fetch(url+path,{headers:{Authorization:`Bearer ${a.token}`},signal:abort.signal}),'HTTP');return{status:r.status,body:await bounded(r.json(),'JSON'),cache:r.headers.get('cache-control')};}finally{abort.abort();}
  }
  function path(route:Route,extra='',b=building){return `/v1/tenancy/${route}?buildingId=${b}${extra}`;}
  async function actor(label:string,role='BUILDING_ADMIN'):Promise<Actor>{
    const id=`struct-${suffix}-${label}`,binding=randomUUID(),generation=randomUUID();users.push(id);
    await sqlClient`insert into users(id,name,email,password_hash) values(${id},${label},${id+'@example.invalid'},${hash})`;
    await sqlClient`insert into memberships(user_id,building_id,role) values(${id},${building},${role})`;
    await sqlClient`insert into role_bindings(id,user_id,role_key,active) values(${binding},${id},'PLATFORM_ADMIN',false)`;
    const abort=new AbortController();try{const r=await bounded(fetch(url+'/auth/login',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({email:id+'@example.invalid',password}),signal:abort.signal}),'login');assert.equal(r.status,200);const body=await bounded(r.json(),'login JSON') as {accessToken:string};const sid=decodeJwt(body.accessToken).sid;assert.equal(typeof sid,'string');return{id,token:body.accessToken,sid:sid as string,binding,generation};}finally{abort.abort();}
  }
  const prove=(a:Actor)=>sqlClient`update sessions set mfa_verified_at=now(),mfa_factor_generation=${a.generation} where id=${a.sid}`;
  before(async()=>{
    const [ledger]=await sqlClient`select count(*)::int as count from schema_migrations`;assert.equal(ledger?.count,68);
    hash=await hashPassword(password);await sqlClient`insert into organizations(id,name,slug) values(${org},'Structure',${org})`;
    for(const b of [building,foreign])await sqlClient`insert into buildings(id,organization_id,name,code) values(${b},${org},${b},${b})`;
    for(let i=0;i<3;i++){
      await sqlClient`insert into blocks(id,building_id,code,name,active) values(${ids.blocks[i]!},${building},${'B'+i},${'Block '+i},${i<2})`;
      await sqlClient`insert into units(id,building_id,code,block_id,floor,active) values(${ids.units[i]!},${building},${'U'+i},${i===1?ids.blocks[1]!:null},${i===0?null:-1},${i<2})`;
      await sqlClient`insert into teams(id,building_id,name,active) values(${ids.teams[i]!},${building},${'Team '+i},${i<2})`;
    }
    server=await new Promise<Server>(resolve=>{const s=app.listen(0,'127.0.0.1',()=>resolve(s));});const a=server.address();assert.ok(a&&typeof a!=='string');url=`http://127.0.0.1:${a.port}`;
    manager=await actor('manager');resident=await actor('resident','RESIDENT');satisfied=await actor('satisfied');
    await sqlClient`insert into mfa_factors(user_id,secret_ciphertext,confirmed_at,generation) values(${satisfied.id},decode(repeat('00',40),'hex'),now(),${satisfied.generation})`;

  after(async()=>{
    try{await bounded(new Promise<void>((resolve,reject)=>{if(!server){resolve();return;}server.close(e=>e?reject(e):resolve());server.closeAllConnections();}),'listener close');}

  });
  async function interleave(a:Actor,route:Route,extra:string,change:()=>Promise<unknown>){
    const cap=`${route==='teams'?'teams':'units'}:${extra.includes('includeInactive=true')?'manage':'read'}`;
    const barrier=beforeProjection(cap),abort=new AbortController(),pending=request(a,path(route,extra),abort);void pending.catch(()=>undefined);let failed=false;

  }
  function denied(r:HttpResult,reason:string){assert.equal(r.status,403);assert.equal((r.body as {details?:{reason:string}}).details?.reason,reason);assert.equal('items' in (r.body as object),false);}

    }
  });

  });
  it('keeps inactive management listings distinct from ordinary read capability',async()=>{
