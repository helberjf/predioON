import { sql, type SQL } from "drizzle-orm";
import type { AppTransaction } from "@predioon/db/runtime";

type MfaReadOrder<Row> = (keyof Row & string) | Readonly<{
  key: keyof Row & string;
  direction: "asc" | "desc";
}>;

/** Server-authored SELECTs only. Relation-reading SQL/SD dependencies must be
 * reviewed STABLE/IMMUTABLE reads. Reviewed volatile exceptions are the existing
 * pg_catalog.clock_timestamp() wall-clock scalars in the status of
 * GET /v1/tenancy/invitations and GET /v1/support-access/grants, and in the
 * eligibility filter of GET /v1/support-access/mine. Native clock only, with no
 * internal SQL or effects; relation snapshots do not freeze this wall clock.
 * These exceptions never cover VOLATILE SQL/SD/custom readers or mutators.
 * SQL fragments, sort keys and directions must never come from a client. This is not a
 * parser/sandbox for arbitrary SQL. Selected columns need explicit DTO aliases;
 * JSON bypasses Drizzle column decoders, so callers supply explicit decoders.
 * CASE/empty behavior requires real PostgreSQL regression for each read family.
 */
export async function readMfaRows<Row>(
  tx: AppTransaction,
  query: SQL,
  decode: (value: unknown) => Row,
  orderBy: readonly [MfaReadOrder<Row>, ...MfaReadOrder<Row>[]],
): Promise<Row[]> {
  const alias = sql.identifier("mfa_read_row");
  const order = sql.join(orderBy.map(item => {
    // Preserve the SQL of every existing string-key ASC caller.
    if (typeof item === "string") return sql`${alias}.${sql.identifier(item)}`;
    const key = sql`${alias}.${sql.identifier(item.key)}`;
    if (item.direction === "asc") return sql`${key} asc`;
    if (item.direction === "desc") return sql`${key} desc`;
    throw new Error("Invalid guarded read order direction");
  }), sql`, `);
  const [result] = await tx.execute<{ rows: unknown }>(sql`
    select case when public.app_require_session_mfa() then coalesce((
      select jsonb_agg(to_jsonb(${alias}) order by ${order})
      from (${query}) as ${alias}
    ), '[]'::jsonb) else null end as rows
  `);
  // Guard currently returns true or PMF01/02. An unexpected false/null result
  // cannot become a successful empty response, and no private row is decoded.
  if (!result || !Array.isArray(result.rows)) throw new Error("Invalid guarded read result");
  return result.rows.map(decode);
}

/** Restore the Date semantics of timestamp columns after PostgreSQL JSON. */
export function decodeMfaReadDate(value: string): Date {
  const date = new Date(value);
  if (!Number.isFinite(date.getTime())) throw new Error("Invalid guarded read timestamp");
  return date;
}
