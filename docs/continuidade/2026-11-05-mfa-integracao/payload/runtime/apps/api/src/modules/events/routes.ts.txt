import { Router } from "express";
import { decodeJwt } from "jose";
import { inTenantContext } from "../../auth/middleware.js";
import { verifyAccessToken } from "../../auth/tokens.js";
import { resolveIdentity } from "../../auth/service.js";
import { assertOperationalMfa } from "../../auth/mfa-policy.js";
import { unauthorized } from "../../http/errors.js";
import { subscribe } from "./bus.js";
import { projectTelemetryEvent } from "./telemetry.js";
import { projectAlertEvent } from "./alerts.js";
import { projectDeviceStatusEvent, projectGatewayStatusEvent } from "./equipment.js";
import { projectFeatureEvent } from "./features.js";

export const eventsRouter = Router();

/**
 * EventSource cannot send an Authorization header, so the SSE route also accepts
 * ?access_token=. It is the only route that does, and only short-lived access tokens work.
 */
eventsRouter.get("/stream", async (req, res) => {
  const headerToken = req.header("authorization")?.replace("Bearer ", "");
  const token = headerToken ?? (typeof req.query.access_token === "string" ? req.query.access_token : null);
  if (!token) throw unauthorized("Token ausente");

  const claims = await verifyAccessToken(token);
  if (!claims) throw unauthorized("Token inválido ou expirado");

  const initialIdentity = await resolveIdentity(claims);
  // The client may disconnect while the initial identity query is in flight.
  // Do not install timers/subscriptions or write headers on an ended response.
  if (res.destroyed || res.writableEnded) return;
  assertOperationalMfa(initialIdentity);
  req.auth = { ...initialIdentity, sessionId: claims.sid };

  let closed = false, queued = 0, heartbeatPending = false;
  let delivery: Promise<void> = Promise.resolve();
  let heartbeat: ReturnType<typeof setInterval> | undefined;
  let expiry: ReturnType<typeof setTimeout> | undefined;
  let unsubscribe: () => void = () => {};
  const isOpen = () => !closed && !res.writableEnded && !res.destroyed;
  const close = () => {
    if (closed) return;
    closed = true;
    clearInterval(heartbeat);
    clearTimeout(expiry);
    unsubscribe();
    if (!res.writableEnded && !res.destroyed) res.end();
  };
  // Expiry, overflow and transport closure must not wait for a slow query.
  res.on("close", close);
  const enqueue = (work: () => Promise<void>, settled: () => void) => {
    delivery = delivery.then(async () => {
      if (isOpen()) await work();
    }).catch(close).finally(settled);
  };

  res.writeHead(200, {
    "Content-Type": "text/event-stream",
    "Cache-Control": "no-cache, no-transform",
    Connection: "keep-alive",
    "X-Accel-Buffering": "no",
  });
  res.write(`retry: 5000\n\n`);

  unsubscribe = subscribe((event) => {
    if (!isOpen()) return;
    // The budget includes the active event, but not the coalesced heartbeat.
    if (queued >= 100) { close(); return; }
    queued++;
    enqueue(async () => {
      const identity = await resolveIdentity(claims);
      if (!isOpen()) return;
      assertOperationalMfa(identity);
      req.auth = { ...identity, sessionId: claims.sid };
      if (event.kind === "telemetry") {
        const projected = await inTenantContext(req, tx => projectTelemetryEvent(tx, event));
        // No frame may leave before the transaction has successfully settled.
        if (projected && isOpen()) res.write(`event: telemetry\ndata: ${JSON.stringify(projected)}\n\n`);
        return;
      }
      if (event.kind === "alert") {
        const projected = await inTenantContext(req, tx => projectAlertEvent(tx, event));
        // No frame may leave before the transaction has successfully settled.
        if (projected && isOpen()) res.write(`event: alert\ndata: ${JSON.stringify(projected)}\n\n`);
        return;
      }
      if (event.kind === "device-status") {
        const projected = await inTenantContext(req, tx => projectDeviceStatusEvent(tx, event));
        // No frame may leave before the transaction has successfully settled.
        if (projected && isOpen()) res.write(`event: device-status\ndata: ${JSON.stringify(projected)}\n\n`);
        return;
      }
      if (event.kind === "gateway-status") {
        const projected = await inTenantContext(req, tx => projectGatewayStatusEvent(tx, event));
        // No frame may leave before the transaction has successfully settled.
        if (projected && isOpen()) res.write(`event: gateway-status\ndata: ${JSON.stringify(projected)}\n\n`);
        return;
      }
      if (event.kind === "features-changed") {
        const projected = await inTenantContext(req, tx => projectFeatureEvent(tx, event));
        // No frame may leave before the transaction has successfully settled.
        if (projected && isOpen()) res.write(`event: features-changed\ndata: ${JSON.stringify(projected)}\n\n`);
      }
    }, () => { queued--; });
  });

  // One pending/running heartbeat shares the delivery queue. Slow queries do
  // not accumulate ticks, and a heartbeat never overwrites an event's auth.
  heartbeat = setInterval(() => {
    if (!isOpen() || heartbeatPending) return;
    heartbeatPending = true;
    enqueue(async () => {
      const identity = await resolveIdentity(claims);
      if (!isOpen()) return;
      assertOperationalMfa(identity);
      res.write(`: ping\n\n`);
    }, () => { heartbeatPending = false; });
  }, 25_000);

  expiry = setTimeout(close, Math.max(1, (decodeJwt(token).exp! * 1000) - Date.now()));
});
