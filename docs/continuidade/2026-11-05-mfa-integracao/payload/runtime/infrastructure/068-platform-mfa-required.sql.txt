CREATE FUNCTION public.identity_platform_mfa_required(target_user_id text)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER
SET search_path=pg_catalog,public,pg_temp AS $$
 SELECT EXISTS(SELECT 1 FROM public.users u WHERE u.id=target_user_id AND u.active
   AND (u.is_platform_admin OR public.app_rbac_platform_role_at(
     u.id,'PLATFORM_ADMIN',statement_timestamp())));
$$;

CREATE FUNCTION public.identity_session_mfa_requirement(target_user_id text,target_session_id uuid)
RETURNS text LANGUAGE sql STABLE SECURITY DEFINER
SET search_path=pg_catalog,public,pg_temp AS $$
 SELECT CASE
 WHEN NOT public.identity_platform_mfa_required(target_user_id) THEN 'not-required'
 WHEN NOT EXISTS(SELECT 1 FROM public.mfa_factors f
   WHERE f.user_id=target_user_id AND f.confirmed_at IS NOT NULL) THEN 'enrollment-required'
 WHEN EXISTS(SELECT 1 FROM public.sessions s JOIN public.mfa_factors f ON f.user_id=s.user_id
   WHERE s.id=target_session_id AND s.user_id=target_user_id
     AND s.revoked_at IS NULL AND s.expires_at>statement_timestamp()
     AND f.confirmed_at IS NOT NULL AND s.mfa_verified_at IS NOT NULL
     AND s.mfa_factor_generation=f.generation) THEN 'satisfied'
 ELSE 'verification-required' END;
$$;

-- New narrow app entrypoint, deliberately distinct from identity-only helpers.
-- Reads no client supplied actor argument; uses the HTTP transaction context.
-- This is not protection against an arbitrary SQL actor allowed to forge GUCs.
CREATE FUNCTION public.app_require_session_mfa()
RETURNS boolean LANGUAGE plpgsql STABLE SECURITY DEFINER
SET search_path=pg_catalog,public,pg_temp AS $$
DECLARE actor text := nullif(current_setting('app.user_id',true),'');
        sid_text text := nullif(current_setting('app.session_id',true),'');
        sid uuid; requirement text;
BEGIN
  -- No SID bypass: administrative SQL contexts without one remain restricted.
  -- Non-admin users keep optional MFA. Service roles do not use these policies.
  IF NOT public.identity_platform_mfa_required(actor) THEN RETURN true; END IF;
  IF sid_text IS NOT NULL THEN
    BEGIN sid := sid_text::uuid;
    EXCEPTION WHEN invalid_text_representation THEN sid := NULL;
    END;
  END IF;
  requirement := public.identity_session_mfa_requirement(actor,sid);
  IF requirement='enrollment-required' THEN
    RAISE EXCEPTION 'MFA_ENROLLMENT_REQUIRED' USING ERRCODE='PMF01';
  ELSIF requirement='verification-required' THEN
    RAISE EXCEPTION 'MFA_VERIFICATION_REQUIRED' USING ERRCODE='PMF02';
  END IF;
  RETURN true;
END;
$$;

-- Existing app-only STABLE SECURITY DEFINER projections bypass app table RLS.
-- Their original bodies/capability checks/payloads are copied from024 and058;
-- only the first MFA PERFORM, qualified function name and fixed path are added.
-- clock_timestamp()/Timescale reads in058 retain their historical semantics.
CREATE OR REPLACE FUNCTION public.app_overview_platform_health()
RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE result jsonb; directory_allowed boolean;
BEGIN
 PERFORM public.app_require_session_mfa();
 IF NOT app_has_global_capability('platform:read-health') THEN
   RAISE EXCEPTION 'Health capability required' USING ERRCODE='42501';
 END IF;
 directory_allowed := app_has_global_capability('buildings:read');
 WITH active_buildings AS MATERIALIZED (
   SELECT b.id,b.name,b.code,o.name organization_name FROM buildings b
   JOIN organizations o ON o.id=b.organization_id AND o.active WHERE b.active
 ), valid_gateways AS MATERIALIZED (
   SELECT g.id,g.building_id,g.status FROM gateways g JOIN active_buildings b ON b.id=g.building_id
 ), valid_devices AS MATERIALIZED (
   SELECT d.id,d.building_id,d.type,d.status FROM devices d JOIN active_buildings b ON b.id=d.building_id
   WHERE app_equipment_gateway_belongs(d.building_id,d.gateway_id)
 ), valid_alerts AS MATERIALIZED (
   SELECT a.building_id,a.severity FROM alerts a JOIN active_buildings b ON b.id=a.building_id
   WHERE a.status<>'RESOLVED'
     AND (a.device_id IS NULL OR EXISTS(SELECT 1 FROM devices d WHERE d.id=a.device_id AND d.building_id=a.building_id))
     AND (a.gateway_id IS NULL OR EXISTS(SELECT 1 FROM gateways g WHERE g.id=a.gateway_id AND g.building_id=a.building_id))
     AND (a.rule_id IS NULL OR EXISTS(SELECT 1 FROM alert_rules r WHERE r.id=a.rule_id AND r.building_id=a.building_id
       AND (r.device_id IS NULL OR (r.device_id=a.device_id AND EXISTS(SELECT 1 FROM devices d WHERE d.id=r.device_id AND d.building_id=a.building_id)))))
 ), device_totals AS (
   SELECT building_id,count(*) devices,count(*) FILTER(WHERE status='ONLINE') devices_online FROM valid_devices GROUP BY building_id
 ), gateway_totals AS (
   SELECT building_id,count(*) gateways,count(*) FILTER(WHERE status='ONLINE') gateways_online FROM valid_gateways GROUP BY building_id
 ), alert_totals AS (
   SELECT building_id,count(*) open_alerts,count(*) FILTER(WHERE severity IN ('HIGH','CRITICAL')) critical_alerts FROM valid_alerts GROUP BY building_id
 ), category_totals AS (
   SELECT type::text type,count(*) devices,count(*) FILTER(WHERE status='ONLINE') devices_online FROM valid_devices GROUP BY type
 ), building_category_totals AS (
   SELECT building_id,type::text type,count(*) devices,count(*) FILTER(WHERE status='ONLINE') devices_online FROM valid_devices GROUP BY building_id,type
 )
 SELECT jsonb_build_object(
   'counts',jsonb_build_object('organizations',(SELECT count(*) FROM organizations WHERE active),'buildings',(SELECT count(*) FROM active_buildings),
     'users',(SELECT count(*) FROM users WHERE active),'devices',(SELECT count(*) FROM valid_devices),'devices_online',(SELECT count(*) FROM valid_devices WHERE status='ONLINE'),
     'gateways',(SELECT count(*) FROM valid_gateways),'gateways_online',(SELECT count(*) FROM valid_gateways WHERE status='ONLINE'),
     'open_alerts',(SELECT count(*) FROM valid_alerts),'critical_alerts',(SELECT count(*) FROM valid_alerts WHERE severity IN ('HIGH','CRITICAL'))),
   'categories',coalesce((SELECT jsonb_agg(to_jsonb(c) ORDER BY c.type) FROM category_totals c),'[]'::jsonb),
   'directoryAvailability',CASE WHEN directory_allowed THEN 'available' ELSE 'unavailable' END,
   'buildings',CASE WHEN directory_allowed THEN coalesce((SELECT jsonb_agg(jsonb_build_object(
     'id',b.id,'name',b.name,'code',b.code,'organization_name',b.organization_name,
     'devices',coalesce(d.devices,0),'devices_online',coalesce(d.devices_online,0),
     'gateways',coalesce(g.gateways,0),'gateways_online',coalesce(g.gateways_online,0),
     'open_alerts',coalesce(a.open_alerts,0),'critical_alerts',coalesce(a.critical_alerts,0),
     'categories',coalesce((SELECT jsonb_agg(jsonb_build_object('type',c.type,'devices',c.devices,'devices_online',c.devices_online) ORDER BY c.type)
       FROM building_category_totals c WHERE c.building_id=b.id),'[]'::jsonb)) ORDER BY coalesce(a.open_alerts,0) DESC,b.name,b.id)
     FROM active_buildings b LEFT JOIN device_totals d ON d.building_id=b.id
     LEFT JOIN gateway_totals g ON g.building_id=b.id LEFT JOIN alert_totals a ON a.building_id=b.id),'[]'::jsonb) ELSE NULL END
 ) INTO result;
 RETURN result;
END $$;

CREATE OR REPLACE FUNCTION public.app_platform_service_health()
RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE result jsonb;
BEGIN
 PERFORM public.app_require_session_mfa();
 IF NOT app_has_global_capability('platform:read-health') THEN
   RAISE EXCEPTION 'Health capability required' USING ERRCODE='42501';
 END IF;
 SELECT jsonb_build_object(
   'checkedAt', clock_timestamp(),
   'ingest', (SELECT jsonb_build_object(
       'lastReadingAt', (SELECT t.time FROM telemetry t ORDER BY t.time DESC LIMIT 1),
       'lastReceivedAt', (SELECT t.received_at FROM telemetry t ORDER BY t.time DESC LIMIT 1),
       'readingsLastHour', (SELECT count(*) FROM telemetry t WHERE t.time > clock_timestamp()-interval '1 hour' AND t.time <= clock_timestamp()+interval '5 minutes'),
       'gatewaysOnline', (SELECT count(*) FROM gateways g JOIN buildings b ON b.id=g.building_id AND b.active WHERE g.enabled AND g.status='ONLINE'),
       'gatewaysEnabled', (SELECT count(*) FROM gateways g JOIN buildings b ON b.id=g.building_id AND b.active WHERE g.enabled))),
   'notifications', (SELECT jsonb_build_object(
       'waiting', count(*) FILTER (WHERE d.status IN ('pending','retry','inflight')),
       'oldestWaitingSince', min(d.created_at) FILTER (WHERE d.status IN ('pending','retry','inflight')),
       'deliveredLastDay', count(*) FILTER (WHERE d.status='delivered' AND d.updated_at > clock_timestamp()-interval '1 day'),
       'failedLastDay', count(*) FILTER (WHERE d.status='failed' AND d.updated_at > clock_timestamp()-interval '1 day'))
     FROM event_deliveries d),
   'physicalCommands', (SELECT jsonb_build_object(
       'waiting', count(*) FILTER (WHERE c.status IN ('PENDING','SENT') AND c.expires_at > clock_timestamp()),
       'acknowledgedLastDay', count(*) FILTER (WHERE c.status='ACKNOWLEDGED' AND c.created_at > clock_timestamp()-interval '1 day'),
       'failedLastDay', count(*) FILTER (WHERE c.status IN ('FAILED','EXPIRED') AND c.created_at > clock_timestamp()-interval '1 day'))
     FROM gate_commands c),
   'jobs', coalesce((SELECT jsonb_agg(jsonb_build_object(
       'name', CASE WHEN j.proc_name='app_automation_schedule_job' THEN 'automations' WHEN j.proc_name='policy_retention' THEN 'telemetry-retention' ELSE j.proc_name END,
       'scheduled', j.scheduled, 'intervalSeconds', extract(epoch FROM j.schedule_interval)::integer,
       'lastStatus', s.last_run_status, 'lastSuccessAt', s.last_successful_finish, 'failures', coalesce(s.total_failures, 0),
       'nextStartAt', s.next_start) ORDER BY j.job_id)
     FROM timescaledb_information.jobs j LEFT JOIN timescaledb_information.job_stats s ON s.job_id=j.job_id
     WHERE j.proc_name IN ('app_automation_schedule_job','policy_retention')), '[]'::jsonb),
   'database', jsonb_build_object('bytes', pg_database_size(current_database()), 'telemetryBytes', hypertable_size('public.telemetry')))
 INTO result;
 RETURN result;
END $$;

-- Finite telemetry SSE slice: one statement snapshot for the MFA obligation,
-- exact raw/published device authority and feature-state inputs. Event values
-- remain validated NOTIFY inputs; this is not a persisted-sample lookup.
CREATE FUNCTION public.app_mfa_telemetry_event_context(
 target_building_id text, target_device_id text
) RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY INVOKER
SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE result jsonb;
BEGIN
 PERFORM public.app_require_session_mfa();
 WITH raw_scope AS MATERIALIZED (
  SELECT d.* FROM public.app_telemetry_authorized_devices(target_building_id,'telemetry:read') d
  WHERE d.building_id=target_building_id AND d.device_id=target_device_id
 ), published_scope AS MATERIALIZED (
  SELECT d.* FROM public.app_telemetry_authorized_devices(target_building_id,'telemetry:read-published') d
  WHERE d.building_id=target_building_id AND d.device_id=target_device_id
 ), authority AS (
  SELECT EXISTS(SELECT 1 FROM raw_scope) OR EXISTS(SELECT 1 FROM published_scope) AS allowed
 )
 SELECT pg_catalog.jsonb_build_object(
  'raw',(SELECT pg_catalog.to_jsonb(d) FROM raw_scope d LIMIT 1),
  'published',(SELECT pg_catalog.to_jsonb(d) FROM published_scope d LIMIT 1),
  'global',CASE WHEN a.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',g.feature_key,'enabled',g.enabled,'version',g.version) ORDER BY g.feature_key)
   FROM public.global_feature_settings g
  ),'[]'::jsonb) ELSE '[]'::jsonb END,
  'local',CASE WHEN a.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',b.feature_key,'enabled',b.enabled,'version',b.version) ORDER BY b.feature_key)
   FROM public.building_feature_settings b WHERE b.building_id=target_building_id
  ),'[]'::jsonb) ELSE '[]'::jsonb END,
  'runtime',CASE WHEN a.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',r.feature_key,'resumedAt',r.resumed_at) ORDER BY r.feature_key)
   FROM public.feature_runtime r WHERE r.building_id=target_building_id
  ),'[]'::jsonb) ELSE '[]'::jsonb END
 ) INTO result FROM authority a;
 RETURN result;
END $$;

-- Finite alert SSE slice: authorize the exact pair, then reread the persisted
-- DTO under caller RLS and capture feature inputs in the same statement snapshot.
CREATE FUNCTION public.app_mfa_alert_event_context(
 target_building_id text, target_alert_id uuid
) RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY INVOKER
SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE result jsonb;
BEGIN
 PERFORM public.app_require_session_mfa();
 WITH authorized_scope AS MATERIALIZED (
  SELECT c.* FROM public.app_alert_authorized_contexts(target_building_id,target_alert_id) c
  WHERE c.building_id=target_building_id AND c.alert_id=target_alert_id
 ), stored_alert AS MATERIALIZED (
  SELECT a.building_id,a.id,a.device_id,a.gateway_id,a.severity,a.type,a.message,a.status,a.triggered_at
  FROM public.alerts a
  WHERE a.building_id=target_building_id AND a.id=target_alert_id
   AND EXISTS(SELECT 1 FROM authorized_scope c WHERE c.building_id=a.building_id AND c.alert_id=a.id)
 ), authority AS (
  SELECT EXISTS(SELECT 1 FROM stored_alert) AS allowed
 )
 SELECT pg_catalog.jsonb_build_object(
  'authorized',CASE WHEN authority.allowed THEN (SELECT pg_catalog.to_jsonb(c) FROM authorized_scope c LIMIT 1) ELSE NULL END,
  'stored',(SELECT pg_catalog.jsonb_build_object(
   'buildingId',a.building_id,'alertId',a.id,'deviceId',a.device_id,'gatewayId',a.gateway_id,
   'severity',a.severity,'type',a.type,'message',a.message,'status',a.status,'triggeredAt',a.triggered_at
  ) FROM stored_alert a LIMIT 1),
  'global',CASE WHEN authority.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',g.feature_key,'enabled',g.enabled,'version',g.version) ORDER BY g.feature_key)
   FROM public.global_feature_settings g
  ),'[]'::jsonb) ELSE '[]'::jsonb END,
  'local',CASE WHEN authority.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',b.feature_key,'enabled',b.enabled,'version',b.version) ORDER BY b.feature_key)
   FROM public.building_feature_settings b WHERE b.building_id=target_building_id
  ),'[]'::jsonb) ELSE '[]'::jsonb END,
  'runtime',CASE WHEN authority.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',r.feature_key,'resumedAt',r.resumed_at) ORDER BY r.feature_key)
   FROM public.feature_runtime r WHERE r.building_id=target_building_id
  ),'[]'::jsonb) ELSE '[]'::jsonb END
 ) INTO result FROM authority;
 RETURN result;
END $$;

-- Finite device-status slice. Capture persisted status, exact classification
-- and feature inputs together; current status uses ANY enabled feature without
-- observation freshness. A separate guarded statement retains the existing
-- post-context authority refresh for support-window expiry.
CREATE FUNCTION public.app_mfa_device_event_context(
 target_building_id text, target_device_id text
) RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY INVOKER
SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE result jsonb;
BEGIN
 PERFORM public.app_require_session_mfa();
 WITH stored_device AS MATERIALIZED (
  SELECT d.building_id,d.id,d.status FROM public.devices d
  WHERE d.building_id=target_building_id AND d.id=target_device_id
 ), classification AS MATERIALIZED (
  SELECT c.* FROM stored_device d
  CROSS JOIN LATERAL public.app_equipment_device_classification(d.building_id,d.id) c
 ), authority AS (
  SELECT EXISTS(SELECT 1 FROM stored_device) AND EXISTS(SELECT 1 FROM classification) AS allowed
 )
 SELECT pg_catalog.jsonb_build_object(
  'stored',CASE WHEN a.allowed THEN (SELECT pg_catalog.jsonb_build_object(
   'buildingId',d.building_id,'deviceId',d.id,'status',d.status
  ) FROM stored_device d LIMIT 1) ELSE NULL END,
  'classification',CASE WHEN a.allowed THEN (SELECT pg_catalog.to_jsonb(c) FROM classification c LIMIT 1) ELSE NULL END,
  'global',CASE WHEN a.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',g.feature_key,'enabled',g.enabled,'version',g.version) ORDER BY g.feature_key)
   FROM public.global_feature_settings g
  ),'[]'::jsonb) ELSE '[]'::jsonb END,
  'local',CASE WHEN a.allowed THEN coalesce((
   SELECT pg_catalog.jsonb_agg(pg_catalog.jsonb_build_object(
    'key',b.feature_key,'enabled',b.enabled,'version',b.version) ORDER BY b.feature_key)
   FROM public.building_feature_settings b WHERE b.building_id=target_building_id
  ),'[]'::jsonb) ELSE '[]'::jsonb END
 ) INTO result FROM authority a;
 RETURN result;
END $$;

-- This additional fresh observation is intentional, not a replacement for the
-- context guard. Keep MFA and the existing exact capability in this statement.
CREATE FUNCTION public.app_mfa_device_event_allowed(
 target_building_id text, target_device_id text
) RETURNS boolean LANGUAGE plpgsql STABLE SECURITY INVOKER
SET search_path = pg_catalog, public, pg_temp AS $$
BEGIN
 PERFORM public.app_require_session_mfa();
 RETURN public.app_device_has_capability(target_building_id,target_device_id,'devices:read');
END $$;

-- Gateway-status has one persisted read and no feature/classification phase.
-- Guard before target lookup even for NULL, no-match or RLS-hidden gateways.
-- Keep the exact pair and existing capability policies under caller privileges.
CREATE FUNCTION public.app_mfa_gateway_event_context(
 target_building_id text, target_gateway_id text
) RETURNS jsonb LANGUAGE plpgsql STABLE SECURITY INVOKER
CALLED ON NULL INPUT SET search_path = pg_catalog, public, pg_temp AS $$
DECLARE result jsonb;
BEGIN
 PERFORM public.app_require_session_mfa();
 SELECT jsonb_build_object('stored',(
   SELECT jsonb_build_object('buildingId',g.building_id,'gatewayId',g.id,'status',g.status)
   FROM public.gateways g
   WHERE g.building_id=target_building_id AND g.id=target_gateway_id
 )) INTO result;
 RETURN result;
END $$;

-- Close every touched/new entrypoint, including installation-specific default
-- function grants. No identity-only helper is granted to app or service roles.
DO $mfa_v3_acl$
DECLARE controlled_owner name; signature text; grant_row record;
 signatures text[]:=ARRAY[
   'public.identity_platform_mfa_required(text)',
   'public.identity_session_mfa_requirement(text,uuid)',
   'public.app_require_session_mfa()',
   'public.app_list_mfa_organizations()',
   'public.app_list_mfa_devices(text)',
   'public.app_list_mfa_gateways()',
   'public.app_mfa_platform_authorization(text[])',
   'public.app_mfa_scope_authorization(text,text,text,text[])',
   'public.app_overview_platform_health()',
   'public.app_platform_service_health()',
   'public.app_mfa_feature_event_allowed(text)',
   'public.app_mfa_telemetry_event_context(text,text)',
   'public.app_mfa_alert_event_context(text,uuid)',
   'public.app_mfa_device_event_context(text,text)',
   'public.app_mfa_device_event_allowed(text,text)',
   'public.app_mfa_gateway_event_context(text,text)'
 ];
BEGIN
 SELECT pg_catalog.pg_get_userbyid(p.proowner) INTO STRICT controlled_owner
 FROM pg_catalog.pg_proc p
 WHERE p.oid='public.app_rbac_platform_role_at(text,text,timestamptz)'::regprocedure;
 IF controlled_owner IN ('predioon_app','predioon_identity','predioon_broker_auth',
   'predioon_notifications','predioon_ingest','predioon_access_service') THEN
  RAISE EXCEPTION 'Authorization owner must not be a runtime role';
 END IF;
 FOREACH signature IN ARRAY signatures LOOP
  EXECUTE format('ALTER FUNCTION %s OWNER TO %I',signature,controlled_owner);
  FOR grant_row IN
   SELECT DISTINCT a.grantee,pg_catalog.pg_get_userbyid(a.grantee) AS grantee_name
   FROM pg_catalog.pg_proc p
   CROSS JOIN LATERAL pg_catalog.aclexplode(coalesce(p.proacl,pg_catalog.acldefault('f',p.proowner))) a
   WHERE p.oid=signature::regprocedure AND a.grantee<>p.proowner
  LOOP
   IF grant_row.grantee=0 THEN
    EXECUTE format('REVOKE ALL ON FUNCTION %s FROM PUBLIC',signature);
   ELSE
    EXECUTE format('REVOKE ALL ON FUNCTION %s FROM %I',signature,grant_row.grantee_name);
   END IF;
  END LOOP;
 END LOOP;
END;
