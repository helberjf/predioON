import { randomBytes } from "node:crypto";
import { Router } from "express";
import { eq, sql } from "drizzle-orm";
import { z } from "zod";
import { users } from "@predioon/db/runtime";
import { identityDb as db } from "@predioon/db/identity";
import { config } from "../config.js";
import { badRequest, conflict, unauthorized, HttpError } from "../http/errors.js";
import { validateBody } from "../http/validate.js";
import { admitPasswordChange } from "./login-budget.js";
import { authenticateSession, requireOperationalSession, currentAuth } from "./middleware.js";
import { consumeMfaCode, factorOf, newRecoveryCode, open, RECOVERY_CODES, recoveryHash, seal, secretKey, lockMfaFamily, lockCurrentFactor, persistMfaProof, type MfaTransaction } from "./mfa-factor.js";
import { verifyPassword } from "./passwords.js";
import { base32Encode, matchTotp, otpauthUri } from "./totp.js";

async function requirePassword(userId: string, password: string): Promise<{ email: string; passwordHash: string | null }> {
  const [account] = await db.select({ email: users.email, active: users.active, passwordHash: users.passwordHash }).from(users).where(eq(users.id, userId)).limit(1);
  if (!await verifyPassword(password, account?.passwordHash ?? null) || !account?.active) throw badRequest("A senha atual está incorreta");
  return { email: account.email, passwordHash: account.passwordHash };
}

async function lockAccount(tx: MfaTransaction, userId: string, expectedPasswordHash?: string | null): Promise<void> {
  const [locked] = await tx.execute<{ active: boolean }>(sql`select identity_lock_account_active(${userId}) as active`);
  if (!locked?.active) throw unauthorized("Usuário inativo ou inexistente");
  if (expectedPasswordHash !== undefined) {
    const [current] = await tx.select({ passwordHash: users.passwordHash }).from(users).where(eq(users.id, userId)).limit(1);
    if (current?.passwordHash !== expectedPasswordHash) throw badRequest("A senha atual está incorreta");
  }
}

async function denyRequiredMfaDisable(tx: Pick<MfaTransaction, "execute">, userId: string): Promise<void> {
  const [state] = await tx.execute<{ required: boolean }>(sql`select public.identity_platform_mfa_required(${userId}) as required`);
  if (!state || state.required) throw new HttpError(403, "A administração da plataforma exige verificação em duas etapas.", { reason: "MFA_REQUIRED" });
}

const password = z.string().min(1).refine(value => Buffer.byteLength(value, "utf8") <= 1024, "Senha excede 1024 bytes");
const code = z.string().trim().min(6).max(16);
const Enroll = z.object({ currentPassword: password }).strict();
const Confirm = z.object({ code }).strict();
const Disable = z.object({ currentPassword: password, code: code.optional() }).strict();

export const mfaRouter = Router();
mfaRouter.use(authenticateSession);

mfaRouter.get("/", async (req, res) => {
  const userId = currentAuth(req).userId;
  const factor = await factorOf(userId);
  const [remaining] = await db.execute<{ count: number }>(sql`select count(*)::int as count from mfa_recovery_codes where user_id=${userId} and used_at is null`);
  res.json({ enabled: Boolean(factor?.confirmed_at), pending: Boolean(factor && !factor.confirmed_at), recoveryCodesRemaining: factor?.confirmed_at ? remaining!.count : 0,
    available: Boolean(config.mfaEncryptionKey) });
});

/** Starts (or restarts) enrolment. Nothing is enforced until the first code is confirmed. */
mfaRouter.post("/enroll", validateBody(Enroll), admitPasswordChange, async (req, res) => {
  const userId = currentAuth(req).userId;
  secretKey();
  const { email, passwordHash } = await requirePassword(userId, (req.body as z.infer<typeof Enroll>).currentPassword);
  const secret = randomBytes(20);
  // A pending enrolment is replaced; a confirmed factor is never overwritten.
  const stored = await db.transaction(async tx => {
    await lockAccount(tx, userId, passwordHash);
    await lockMfaFamily(tx, userId, currentAuth(req).sessionId);
    await tx.execute(sql`delete from mfa_factors where user_id=${userId} and confirmed_at is null`);
    return tx.execute(sql`insert into mfa_factors(user_id,secret_ciphertext) values(${userId},${seal(secret, userId)}) on conflict (user_id) do nothing returning user_id`);
  });
  if (!stored.length) throw conflict("A verificação em duas etapas já está ativa nesta conta.");
  res.json({ secret: base32Encode(secret), otpauthUri: otpauthUri("Prédio ON", email, secret) });
});

mfaRouter.post("/confirm", validateBody(Confirm), admitPasswordChange, async (req, res) => {
  const userId = currentAuth(req).userId;
  const factor = await factorOf(userId);
  if (!factor) throw conflict("Inicie a configuração antes de confirmar o código.");
  if (factor.confirmed_at) throw conflict("A verificação em duas etapas já está ativa nesta conta.");
  const step = matchTotp(open(Buffer.from(factor.secret_ciphertext), userId), (req.body as z.infer<typeof Confirm>).code, Date.now());
  if (step === null) throw badRequest("Código de verificação inválido. Confira o horário do aparelho e tente o código atual.");
  const codes = Array.from({ length: RECOVERY_CODES }, newRecoveryCode);
  const confirmed = await db.transaction(async tx => {
    await lockAccount(tx, userId);
    await lockMfaFamily(tx, userId, currentAuth(req).sessionId);
    const rows = await tx.execute(sql`update mfa_factors set confirmed_at=clock_timestamp(), last_used_step=${step} where user_id=${userId} and generation=${factor.generation}
      and secret_ciphertext=${factor.secret_ciphertext} and confirmed_at is null returning user_id`);
    if (!rows.length) return false;
    await tx.execute(sql`delete from mfa_recovery_codes where user_id=${userId}`);
    for (const item of codes) await tx.execute(sql`insert into mfa_recovery_codes(user_id,code_hash) values(${userId},${recoveryHash(item)})`);
    // Enrollment proves this factor in this family, not a fresh password (055).
    await persistMfaProof(tx, userId, currentAuth(req).sessionId, { generation: factor.generation });
    return true;
  });
  if (!confirmed) throw conflict("A verificação em duas etapas já está ativa nesta conta.");
  res.json({ recoveryCodes: codes });
});

/** Turning the factor off needs both the password and a valid code. */
mfaRouter.post("/disable", requireOperationalSession, validateBody(Disable), admitPasswordChange, async (req, res) => {
  const userId = currentAuth(req).userId;
  const input = req.body as z.infer<typeof Disable>;
  const account = await requirePassword(userId, input.currentPassword);
  await denyRequiredMfaDisable(db, userId);
  const factor = await factorOf(userId);
  const proof = factor?.confirmed_at && input.code ? await consumeMfaCode(userId, input.code) : null;
  if (factor?.confirmed_at && !proof) throw unauthorized("Código de verificação inválido");
  await db.transaction(async tx => {
    await lockAccount(tx, userId, account.passwordHash);
    await lockMfaFamily(tx, userId, currentAuth(req).sessionId);
    // Current union of legacy flag and021 authority. Concurrent promotion at
    // the operation snapshot remains a separate authorization-boundary task.
    const current = await lockCurrentFactor(tx, userId);
    await denyRequiredMfaDisable(tx, userId);
    if (!current && !factor) return;
    if (!current || !factor || current.generation !== factor.generation
      || (current.confirmed_at && proof?.generation !== current.generation)) {
      throw conflict("O fator foi alterado. Confirme o fator atual.");
    }
    await tx.execute(sql`delete from mfa_factors where user_id=${userId} and generation=${current.generation}`);
  });
  res.status(204).end();
});
