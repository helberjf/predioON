import { Router, type Request } from "express";
import { and, asc, eq, sql } from "drizzle-orm";
import { z } from "zod";
import { memberships, users, type AppTransaction } from "@predioon/db/runtime";
import { assertCapability, assertGlobalCapability, currentAuth, inTenantContext } from "../../auth/middleware.js";
import { badRequest, conflict, forbidden, notFound, pgErrorCode } from "../../http/errors.js";
import { generateId } from "../../http/ids.js";
import { query, validateBody, validateQuery } from "../../http/validate.js";
import { hashPassword } from "../../auth/passwords.js";
import { recordAudit } from "../audit/repo.js";
import { requireRecentVerification } from "../../auth/verification.js";
import { param } from "../../http/params.js";
import { decodeMfaReadDate, readMfaRows } from "../../auth/mfa-read.js";

export const usersRouter = Router();

const ListQuerySchema = z.object({ buildingId: z.string().optional() });

const CreateSchema = z.object({
  name: z.string().min(2).max(120),
  email: z.string().email().max(160),
  password: z.string().min(8).max(200),
  phone: z.string().max(40).optional(),
  isPlatformAdmin: z.boolean().default(false),
});

const MembershipSchema = z.object({
  userId: z.string().min(1),
  buildingId: z.string().min(1),
  role: z.enum(["BUILDING_ADMIN", "RESIDENT"]),
  unit: z.string().max(40).optional(),
});

const publicFields = {
  id: users.id,
  name: users.name,
  email: users.email,
  phone: users.phone,
  isPlatformAdmin: users.isPlatformAdmin,
  active: users.active,
  createdAt: users.createdAt,
};

// SELECT-only aliases: publicFields keeps its existing write RETURNING mapping.
const UserReadRow = z.object({
  id: z.string(), name: z.string(), email: z.string(), phone: z.string().nullable(),
  isPlatformAdmin: z.boolean(), active: z.boolean(),
  createdAt: z.string().transform(decodeMfaReadDate),
}).strict();
const userReadFields = {
  id: sql`${users.id}`.as("id"), name: sql`${users.name}`.as("name"),
  email: sql`${users.email}`.as("email"), phone: sql`${users.phone}`.as("phone"),
  isPlatformAdmin: sql`${users.isPlatformAdmin}`.as("isPlatformAdmin"),
  active: sql`${users.active}`.as("active"), createdAt: sql`${users.createdAt}`.as("createdAt"),
};

/**
 * Accounts and legacy building memberships (054): live capabilities decide,
 * never the legacy role. The platform RBAC administrator (rbac:manage) sees and
 * creates accounts; a building's managers list and link its people
 * (memberships:read/manage), limited by the role delegation rule in the database.
 */
async function inUsers<T>(req: Request, run: (tx: AppTransaction) => Promise<T>): Promise<T> {
  try { return await inTenantContext(req, run); }
  catch (error) {
    const code = pgErrorCode(error);
    if (code === "42501") throw forbidden("Sem a capacidade necessária para este vínculo");
    if (code === "23503") throw badRequest("Pessoa ou condomínio inexistente");
    throw error;
  }
}
async function platformAdministrator(tx: AppTransaction): Promise<boolean> {
  const [row] = await tx.execute(sql`select app_has_global_capability('rbac:manage') as allowed`);
  return row?.allowed === true;
}
/** Linking people needs the building capability, or the platform RBAC capability. */
async function assertMembershipManager(tx: AppTransaction, buildingId: string): Promise<void> {
  if (!await platformAdministrator(tx)) await assertCapability(tx, "memberships:manage", buildingId);
}

usersRouter.get("/", validateQuery(ListQuerySchema), async (req, res) => {
  const { buildingId } = query<z.infer<typeof ListQuerySchema>>(req);
  const rows = await inUsers(req, async (tx) => {
    if (!buildingId) {
      await assertGlobalCapability(tx, "rbac:manage");
      return readMfaRows(tx,
        tx.select(userReadFields).from(users).orderBy(asc(users.name)).getSQL(),
        value => UserReadRow.parse(value), ["name"],
      );
    }
    if (!await platformAdministrator(tx)) await assertCapability(tx, "memberships:read", buildingId);
    return readMfaRows(tx,
      tx.selectDistinct(userReadFields)
        .from(users)
        .innerJoin(memberships, eq(memberships.userId, users.id))
        .where(eq(memberships.buildingId, buildingId))
        .orderBy(asc(users.name)).getSQL(),
      value => UserReadRow.parse(value), ["name"],
    );
  });

  res.json({ items: rows });
});

usersRouter.post("/", validateBody(CreateSchema), async (req, res) => {
  const input = req.body as z.infer<typeof CreateSchema>;
  const auth = currentAuth(req);
  const passwordHash = await hashPassword(input.password);

  const row = await inUsers(req, async (tx) => {
    await assertGlobalCapability(tx, "rbac:manage");
    await requireRecentVerification(req);
    const [existing] = await readMfaRows(tx,
      tx.select({ id: sql`${users.id}`.as("id") }).from(users)
        .where(eq(users.email, input.email.toLowerCase())).limit(1).getSQL(),
      value => z.object({ id: z.string() }).strict().parse(value), ["id"],
    );
    if (existing) throw conflict("Já existe um usuário com este e-mail");

    // Bind MFA to this INSERT snapshot as well as the duplicate lookup.
    // The scalar envelope consumes the guard even if no row is produced.
    const [result] = await tx.execute<{ rows: unknown }>(sql`
      with user_mfa as materialized (
        select public.app_require_session_mfa() as allowed
      ), created as (
        insert into users(id,name,email,phone,is_platform_admin,password_hash)
        select ${generateId("usr")},${input.name},${input.email.toLowerCase()},${input.phone ?? null},${input.isPlatformAdmin},${passwordHash}
        from user_mfa where user_mfa.allowed
        returning ${sql.join(Object.entries(publicFields).map(([alias, column]) => sql`${column} as ${sql.identifier(alias)}`), sql`, `)}
      )
      select case when user_mfa.allowed then coalesce((
        select jsonb_agg(to_jsonb(created)) from created
      ), '[]'::jsonb) else null end as rows from user_mfa
    `);
    if (!result || !Array.isArray(result.rows) || result.rows.length !== 1) throw new Error("Invalid guarded account creation");
    const created = UserReadRow.parse(result.rows[0]);

    await recordAudit(tx, req, {
      userId: auth.userId,
      action: "USER_CREATED",
      resourceType: "user",
      resourceId: created.id,
    });
    // A later audit/trigger statement must not commit a now-restricted action.
    const [outcome] = await tx.execute<{ allowed: unknown }>(sql`select public.app_require_session_mfa() as allowed`);
    if (outcome?.allowed !== true) throw new Error("Invalid guarded account outcome");
    return created;
  });

  res.status(201).json(row);
});

usersRouter.post("/memberships", validateBody(MembershipSchema), async (req, res) => {
  const input = req.body as z.infer<typeof MembershipSchema>;
  const auth = currentAuth(req);

  const row = await inUsers(req, async (tx) => {
    // The database also refuses granting a role whose capabilities the actor lacks.
    await assertMembershipManager(tx, input.buildingId);
    // Making someone a manager is privileged; linking a resident is routine.
    if (input.role === "BUILDING_ADMIN") await requireRecentVerification(req);
    const [created] = await tx
      .insert(memberships)
      .values(input)
      .onConflictDoUpdate({
        target: [memberships.userId, memberships.buildingId],
        set: {
          role: input.role, unit: input.unit ?? null, active: true,
          // Readmission must not retain the revoked grant's end date. An edit
          // to a still-valid or future grant must preserve its access window.
          startsAt: sql`case when not ${memberships.active} or ${memberships.endsAt} <= statement_timestamp() then null else ${memberships.startsAt} end`,
          endsAt: sql`case when not ${memberships.active} or ${memberships.endsAt} <= statement_timestamp() then null else ${memberships.endsAt} end`,
          updatedAt: sql`clock_timestamp()`,
        },
      })
      .returning();

    await recordAudit(tx, req, {
      buildingId: input.buildingId,
      userId: auth.userId,
      action: "MEMBERSHIP_UPSERTED",
      resourceType: "membership",
      resourceId: created!.id,
      metadata: { targetUserId: input.userId, role: input.role },
    });
    return created!;
  });

  res.status(201).json(row);
});

usersRouter.delete("/memberships/:membershipId", async (req, res) => {
  const auth = currentAuth(req);

  const membershipId = z.string().uuid().safeParse(param(req, "membershipId"));
  if (!membershipId.success) throw notFound("Vínculo não encontrado");
  await inUsers(req, async (tx) => {
    const [current] = await tx.select().from(memberships).where(eq(memberships.id, membershipId.data)).limit(1);
    if (!current) throw notFound("Vínculo não encontrado");
    await assertMembershipManager(tx, current.buildingId);

    // Record while the actor still has authority, including self-revocation.
    // Both the audit and the mutation roll back if either operation fails.
    await recordAudit(tx, req, {
      buildingId: current.buildingId,
      userId: auth.userId,
      action: "MEMBERSHIP_REVOKED",
      resourceType: "membership",
      resourceId: current.id,
    });
    const changed = await tx
      .update(memberships)
      .set({ active: false, endsAt: new Date(), updatedAt: new Date() })
      .where(and(eq(memberships.id, current.id)))
      .returning({ id: memberships.id });
    if (changed.length !== 1) throw conflict("O vínculo mudou durante a revogação. Atualize e tente novamente.");
  });

  res.status(204).end();
});
