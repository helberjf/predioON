/** Listing inactive records needs the management capability, not only the read one. */
async function listScope(tx: AppTransaction, q: StructureQuery, read: "units:read" | "teams:read", manage: "units:manage" | "teams:manage"): Promise<boolean> {
  await assertCapability(tx, read, q.buildingId);
  if (q.includeInactive !== "true") return false;
  await assertCapability(tx, manage, q.buildingId);
  return true;
}

function page<T extends { id: string }>(rows: T[], limit: number) {
  const items = rows.slice(0, limit);
  return { items, nextCursor: rows.length > limit ? items.at(-1)!.id : null };
}

/** RLS is enforced by the non-owner connection; constraint errors never expose foreign records. */
async function inTenancy<T>(req: Request, run: (tx: AppTransaction) => Promise<T>): Promise<T> {
  try { return await inTenantContext(req, run); }
  catch (error) {
    const code = pgErrorCode(error);
    if (code === "23503" || code === "23514") throw badRequest("Vínculo ou vigência inválidos para este condomínio");
    if (code === "23505") throw conflict("Este cadastro já existe no condomínio");
    if (code === "42501") throw forbidden();
    throw error;
  }
}

/** Only the three server-authored INSERT SELECTs below call this helper.
 * Each INSERT consumes tenancy_create_mfa; the scalar also consumes it when
 * RETURNING is empty. This DML must not be passed to readMfaRows.
 */
async function insertStructure<Row extends { id: string }>(tx: AppTransaction, statement: SQL, decode: (value: unknown) => Row): Promise<Row> {
  const result = await tx.execute<{ rows: unknown }>(sql`
    with tenancy_create_mfa as materialized (
      select public.app_require_session_mfa() as allowed
    ), created as (
      ${statement}
    )
    select case when tenancy_create_mfa.allowed then coalesce((
      select jsonb_agg(to_jsonb(created)) from created
    ), '[]'::jsonb) else null end as rows from tenancy_create_mfa
  `);
  const rows = result[0]?.rows;
  // These creates have no normal no-op. Unexpected zero/multiple RETURNING
  // rows are an internal failure and roll back, never a successful201.
  if (result.length !== 1 || !Array.isArray(rows) || rows.length !== 1) throw new Error("Invalid guarded structure creation");
  return decode(rows[0]);
}

tenancyRouter.post("/blocks", validateBody(CreateBlock), async (req, res) => {
  const input = req.body as z.infer<typeof CreateBlock>;
  const row = await inTenancy(req, async tx => {
    await assertCapability(tx, "units:manage", input.buildingId);
    const created = await insertStructure(tx, sql`
      insert into public.blocks (building_id, code, name)
      select ${input.buildingId}, ${input.code}, ${input.name}
      from tenancy_create_mfa where tenancy_create_mfa.allowed
      returning id, building_id as "buildingId", code, name, active,
        created_at as "createdAt", updated_at as "updatedAt"
    `, value => BlockCreateRow.parse(value));
    await recordAudit(tx, req, { userId: currentAuth(req).userId, buildingId: input.buildingId, action: "BLOCK_CREATED", resourceType: "block", resourceId: created.id });
    // A new decision after audit also observes changes made by audit triggers.
    await requireStructureOutcomeMfa(tx);
    return created;
  });
  res.status(201).json(row);
});

/** Only these two grant INSERTs consume grant_mfa. Preserve the complete ORM
 * rows and defaults; unlike team renewal, neither operation has a normal no-op.
 */
async function insertGrant<Row extends { id: string }>(tx: AppTransaction, statement: SQL, decode: (value: unknown) => Row): Promise<Row> {
  const result = await tx.execute<{ rows: unknown }>(sql`
    with grant_mfa as materialized (
      select public.app_require_session_mfa() as allowed
    ), granted as (
      ${statement}
    )
    select case when grant_mfa.allowed then coalesce((
      select jsonb_agg(to_jsonb(granted)) from granted
    ), '[]'::jsonb) else null end as rows from grant_mfa
  `);
  const rows = result[0]?.rows;
  if (result.length !== 1 || !Array.isArray(rows) || rows.length !== 1) throw new Error("Invalid guarded tenancy grant");
  return decode(rows[0]);
}

tenancyRouter.post("/team-members", validateBody(CreateTeamMember), async (req, res) => {
  const input = req.body as z.infer<typeof CreateTeamMember>;
  const row = await inTenancy(req, async tx => {
    await assertCapability(tx, "teams:manage", input.buildingId);
    // Consume MFA even when the active-member conflict yields no RETURNING row.
    // Renewal preserves identity and clears omitted validity dates as before.
    const result = await tx.execute<{ rows: unknown }>(sql`
      with team_mfa as materialized (
        select public.app_require_session_mfa() as allowed
      ), changed as (
        insert into public.team_members (building_id, team_id, user_id
          ${input.startsAt !== undefined ? sql`, starts_at` : sql``}
          ${input.endsAt !== undefined ? sql`, ends_at` : sql``})
        select ${input.buildingId}, ${input.teamId}::uuid, ${input.userId}
          ${input.startsAt !== undefined ? sql`, ${input.startsAt?.toISOString() ?? null}::timestamptz` : sql``}
          ${input.endsAt !== undefined ? sql`, ${input.endsAt?.toISOString() ?? null}::timestamptz` : sql``}
        from team_mfa where team_mfa.allowed
        on conflict (team_id, user_id) do update
          set active=true, starts_at=${input.startsAt?.toISOString() ?? null}::timestamptz,
            ends_at=${input.endsAt?.toISOString() ?? null}::timestamptz,
            updated_at=${new Date().toISOString()}::timestamptz
          where team_members.building_id=${input.buildingId}
            and (team_members.active=false or team_members.ends_at <= current_timestamp)
        returning id, building_id as "buildingId", team_id as "teamId", user_id as "userId", active,
          starts_at as "startsAt", ends_at as "endsAt", created_at as "createdAt", updated_at as "updatedAt"
      )
      select case when team_mfa.allowed then coalesce((
        select jsonb_agg(to_jsonb(changed)) from changed
      ), '[]'::jsonb) else null end as rows from team_mfa
    `);
    const rows = result[0]?.rows;
    if (result.length !== 1 || !Array.isArray(rows) || rows.length > 1) throw new Error("Invalid guarded team membership");
    if (rows.length === 0) {
      await requireStructureOutcomeMfa(tx);
      throw conflict("Este integrante já possui vínculo ativo na equipe");
    }
    const created = TeamMemberCreateRow.parse(rows[0]);
    // This separate statement must see the resulting membership. A sibling
    // SELECT inside the write CTE would not see the same post-upsert state.
    const [person] = await tx.execute(sql`select app_tenancy_is_resident(${input.buildingId}, ${input.userId}) as resident`);
    if (person?.resident) throw badRequest("Moradores não podem integrar equipes do condomínio.");
    await recordAudit(tx, req, { userId: currentAuth(req).userId, buildingId: input.buildingId, action: "TEAM_MEMBER_ADDED", resourceType: "team_member", resourceId: created.id,

// This private command family has exactly three server-authored descriptors.
// Neither relation names, projections nor writable column names come from input.
type StructureView = z.infer<typeof BlockReadRow> | z.infer<typeof UnitReadRow> | z.infer<typeof TeamReadRow>;
type StructureCommand = Readonly<{
  table: SQL; projection: SQL; decode: (value: unknown) => StructureView;
  capability: "units:manage" | "teams:manage";
  prefix: "BLOCK" | "UNIT" | "TEAM"; resourceType: "block" | "unit" | "team";
  editColumns: Readonly<Record<string, string>>;
}>;
function structureCommandProjection(columns: Readonly<Record<string, { readonly name: string }>>): SQL {
  return sql.join(Object.entries(columns).map(([key, column]) =>
    sql`${sql.identifier("target")}.${sql.identifier(column.name)} as ${sql.identifier(key)}`), sql`, `);
}
const structureCommands = {
  block: {
    table: sql`public.blocks`,
    projection: structureCommandProjection(blockView),
    decode: (value: unknown) => BlockReadRow.parse(value),
    capability: "units:manage", prefix: "BLOCK", resourceType: "block", editColumns: { code: "code", name: "name" },
  },
  unit: {
    table: sql`public.units`,
    projection: structureCommandProjection(unitView),
    decode: (value: unknown) => UnitReadRow.parse(value),
    capability: "units:manage", prefix: "UNIT", resourceType: "unit", editColumns: { code: "code", blockId: "block_id", floor: "floor" },
  },
  team: {
    table: sql`public.teams`,
    projection: structureCommandProjection(teamView),
    decode: (value: unknown) => TeamReadRow.parse(value),
    capability: "teams:manage", prefix: "TEAM", resourceType: "team", editColumns: { name: "name" },
  },
} satisfies Record<"block" | "unit" | "team", StructureCommand>;
const StructureTargetRow = z.object({ id: z.string(), buildingId: z.string(), active: z.boolean() }).strict();
type StructureTarget = z.infer<typeof StructureTargetRow>;
const UnchangedStructureRow = z.object({
  allowed: z.boolean(), item: z.record(z.string(), z.unknown()).nullable(),
}).strict();

async function findStructureTarget(tx: AppTransaction, command: StructureCommand, recordId: string, activeOnly: boolean): Promise<StructureTarget | undefined> {
  const rows = await readMfaRows(tx, sql`
    select target.id as "id", target.building_id as "buildingId", target.active as "active"
    from ${command.table} as target where target.id = ${recordId}::uuid
      ${activeOnly ? sql`and target.active` : sql``} limit 1
  `, value => StructureTargetRow.parse(value), ["id"]);
  return rows[0];
}

async function requireStructureOutcomeMfa(tx: AppTransaction): Promise<void> {
  const [row] = await tx.execute<{ allowed: unknown }>(sql`select public.app_require_session_mfa() as allowed`);
  if (row?.allowed !== true) throw new Error("Invalid guarded structure outcome");
}

async function lockStructureTarget(tx: AppTransaction, command: StructureCommand, target: StructureTarget, activeOnly: boolean): Promise<StructureTarget> {
  // Keep the initial SELECT visibility check separate: taking this write lock
  // first could hide a resident's visible unit instead of returning its403.
  const rows = await tx.execute(sql`
    select target.id as "id", target.building_id as "buildingId", target.active as "active"
    from ${command.table} as target
    where target.id = ${target.id}::uuid and target.building_id = ${target.buildingId}
      ${activeOnly ? sql`and target.active` : sql``} for update
  `);
  if (rows.length > 1) throw new Error("Invalid locked structure result");
  if (!rows[0]) {
    // A lock wait can end with no visible target. Do not let404 mask MFA.
    await requireStructureOutcomeMfa(tx);
    throw notFound();
  }
  return StructureTargetRow.parse(rows[0]);
}

function structureEditAssignments(command: StructureCommand, input: Record<string, unknown>): SQL {
  const entries = Object.entries(input).filter(([, value]) => value !== undefined);
  if (!entries.length) throw new Error("Invalid validated structure edit");
  const assignments = entries.map(([key, value]) => {
    if (!Object.hasOwn(command.editColumns, key)) throw new Error("Invalid structure edit column");
    return sql`${sql.identifier(command.editColumns[key]!)} = ${value}`;
  });
  // Preserve the existing API Date/millisecond timestamp semantics.
  assignments.push(sql`updated_at = ${new Date().toISOString()}::timestamptz`);
  return sql.join(assignments, sql`, `);
}

async function writeStructureTarget(tx: AppTransaction, command: StructureCommand, target: StructureTarget, assignments: SQL, activeOnly: boolean): Promise<StructureView> {
  // This is a command, not a readMfaRows SELECT. Both the UPDATE and scalar
  // result depend on the guard; zero matches still require an MFA decision.
  // Existing target USING/WITH CHECK policies retain the capability decision.
  const [result] = await tx.execute<{ rows: unknown }>(sql`
    with tenancy_mfa as materialized (
      select public.app_require_session_mfa() as allowed
    ), changed as (
      update ${command.table} as target set ${assignments}
      from tenancy_mfa
      where tenancy_mfa.allowed and target.id = ${target.id}::uuid
        and target.building_id = ${target.buildingId} ${activeOnly ? sql`and target.active` : sql``}
      returning ${command.projection}
    )
    select case when tenancy_mfa.allowed then coalesce((
      select jsonb_agg(to_jsonb(changed)) from changed
    ), '[]'::jsonb) else null end as rows from tenancy_mfa
  `);
  if (!result || !Array.isArray(result.rows) || result.rows.length > 1) throw new Error("Invalid guarded structure write");
  if (!result.rows.length) throw notFound();
  return command.decode(result.rows[0]);
}

async function unchangedStructure(tx: AppTransaction, command: StructureCommand, target: StructureTarget): Promise<StructureView> {
  // A no-op must still require management in the same snapshot as its view;
  // keeping read permission after a manage grant expires is not sufficient.
  const rows = await readMfaRows(tx, sql`
    select public.app_has_capability(${target.buildingId}, ${command.capability}) as allowed,
      (select to_jsonb(structure_view) from (
        select ${command.projection} from ${command.table} as target
        where target.id = ${target.id}::uuid and target.building_id = ${target.buildingId} limit 1
      ) as structure_view) as item
  `, value => UnchangedStructureRow.parse(value), ["allowed"]);
  const row = rows[0];
  if (rows.length !== 1 || !row) throw new Error("Invalid guarded unchanged structure");
  if (!row.allowed) throw forbidden("Sem a capacidade necessária para este recurso");
  if (!row.item) throw notFound();
  return command.decode(row.item);
}

/** Initial read keeps404/403 visibility; the explicit target lock precedes the
 * guarded UPDATE, and audit remains after the write in the same transaction. */
async function edit(req: Request, command: StructureCommand): Promise<StructureView> {
  const parsed = id.safeParse(param(req, "id"));
  if (!parsed.success) throw notFound();
  return inTenancy(req, async tx => {
    const visible = await findStructureTarget(tx, command, parsed.data, true);
    if (!visible) throw notFound();
    await assertCapability(tx, command.capability, visible.buildingId);
    const current = await lockStructureTarget(tx, command, visible, true);
    const updated = await writeStructureTarget(tx, command, current, structureEditAssignments(command, req.body as Record<string, unknown>), true);
    await recordAudit(tx, req, { userId: currentAuth(req).userId, buildingId: current.buildingId, action: `${command.prefix}_UPDATED`, resourceType: command.resourceType, resourceId: parsed.data,
      metadata: { fields: Object.keys(req.body as object).sort() } });
    // Fresh MFA after audit/trigger execution; do not recheck a capability
    // that a legitimate status command may itself remove.
    await requireStructureOutcomeMfa(tx);
    return updated;
  });
}

// Deactivation keeps the record and its history. Audit before the write lets
// an actor legitimately deactivate the team that grants their own management.
const UpdateStatus = z.object({ active: z.boolean() }).strict();
async function setStatus(req: Request, command: StructureCommand): Promise<StructureView> {
  const parsed = id.safeParse(param(req, "id"));
  if (!parsed.success) throw notFound();
  const { active } = req.body as z.infer<typeof UpdateStatus>;
  return inTenancy(req, async tx => {
    const visible = await findStructureTarget(tx, command, parsed.data, false);
    if (!visible) throw notFound();
    await assertCapability(tx, command.capability, visible.buildingId);
    const current = await lockStructureTarget(tx, command, visible, false);
    if (current.active === active) return unchangedStructure(tx, command, current);
    await recordAudit(tx, req, { userId: currentAuth(req).userId, buildingId: current.buildingId, action: `${command.prefix}_${active ? "REACTIVATED" : "DEACTIVATED"}`, resourceType: command.resourceType, resourceId: parsed.data });
    const updated = await writeStructureTarget(tx, command, current,
      sql`active = ${active}, updated_at = ${new Date().toISOString()}::timestamptz`, false);
    await requireStructureOutcomeMfa(tx);
    return updated;
  });
}

/** Only these two manager commands supply the predicate. Preserve the original
 * SELECT visibility and closed204 semantics; a write lock would hide closed
 * invitations under the UPDATE policy. An empty UPDATE still requires MFA. */
async function revokeInvitationRows(tx: AppTransaction, actor: string, predicate: SQL): Promise<Array<{ id: string }>> {
  const [result] = await tx.execute<{ rows: unknown }>(sql`
    with invitation_mfa as materialized (
      select public.app_require_session_mfa() as allowed
    ), changed as (
      update invitations set revoked_at=clock_timestamp(), revoked_by=${actor}
      from invitation_mfa
      where invitation_mfa.allowed and ${predicate}
        and accepted_at is null and revoked_at is null
      returning id
    )
    select case when invitation_mfa.allowed then coalesce((
      select jsonb_agg(to_jsonb(changed)) from changed
    ), '[]'::jsonb) else null end as rows from invitation_mfa
  `);
  if (!result || !Array.isArray(result.rows) || result.rows.length > 1) throw new Error("Invalid guarded invitation revocation");
  return result.rows.map(value => InvitationRevokedRow.parse(value));
}

// Grants are revoked rather than deleted so their audit references remain useful.
// These three descriptors are server-authored; no relation or capability is input.
type RevocationCommand = Readonly<{
  path: "unit-memberships" | "team-members" | "role-bindings";
  table: SQL; capability: "memberships:manage" | "teams:manage";
  action: "UNIT_MEMBERSHIP_REVOKED" | "TEAM_MEMBER_REVOKED" | "ROLE_BINDING_REVOKED";
}>;
const revocationCommands = [
  { path: "unit-memberships", table: sql`public.unit_memberships`, capability: "memberships:manage", action: "UNIT_MEMBERSHIP_REVOKED" },
  { path: "team-members", table: sql`public.team_members`, capability: "teams:manage", action: "TEAM_MEMBER_REVOKED" },
  { path: "role-bindings", table: sql`public.role_bindings`, capability: "memberships:manage", action: "ROLE_BINDING_REVOKED" },
] as const satisfies readonly RevocationCommand[];
const RevocationTargetRow = z.object({
  id: z.string(), buildingId: z.string().nullable(), active: z.boolean(), allowed: z.boolean(),
}).strict();
type RevocationTarget = z.infer<typeof RevocationTargetRow>;

async function requireRevocationOutcomeMfa(tx: AppTransaction): Promise<void> {
  const [row] = await tx.execute<{ allowed: unknown }>(sql`select public.app_require_session_mfa() as allowed`);
  if (row?.allowed !== true) throw new Error("Invalid guarded revocation outcome");
}

async function lockRevocationTarget(tx: AppTransaction, command: RevocationCommand, recordId: string): Promise<RevocationTarget | undefined> {
  // Keep FOR UPDATE and its write-policy visibility. A plain SELECT here
  // could expose a reader's row and change the original hidden-target404.
  // Both the lock query and scalar empty result depend on the MFA decision.
  // This statement's snapshot may predate a lock wait; callers recheck after it.
  const [result] = await tx.execute<{ rows: unknown }>(sql`
    with revocation_mfa as materialized (
      select public.app_require_session_mfa() as allowed
    ), locked as materialized (
      select target.id as "id", target.building_id as "buildingId", target.active as "active",
        case when target.building_id is not null
          then public.app_has_capability(target.building_id, ${command.capability}) else false end as "allowed"
      from ${command.table} as target cross join revocation_mfa
      where revocation_mfa.allowed and target.id = ${recordId}::uuid
      for update of target
    )
    select case when revocation_mfa.allowed then coalesce((
      select jsonb_agg(to_jsonb(locked)) from locked
    ), '[]'::jsonb) else null end as rows from revocation_mfa
  `);
  if (!result || !Array.isArray(result.rows) || result.rows.length > 1) throw new Error("Invalid guarded revocation target");
  return result.rows.length ? RevocationTargetRow.parse(result.rows[0]) : undefined;
}

async function writeRevocation(tx: AppTransaction, command: RevocationCommand, target: RevocationTarget): Promise<void> {
  // The same-statement guard protects the UPDATE under its existing RLS and
  // the empty outcome. Do not pass this DML through the read-only helper.
  const [result] = await tx.execute<{ rows: unknown }>(sql`
    with revocation_mfa as materialized (
      select public.app_require_session_mfa() as allowed
    ), changed as (
      update ${command.table} as target
      set active = false, updated_at = ${new Date().toISOString()}::timestamptz
      from revocation_mfa
      where revocation_mfa.allowed and target.id = ${target.id}::uuid
        and target.building_id = ${target.buildingId}
      returning target.id as "id"
    )
    select case when revocation_mfa.allowed then coalesce((
      select jsonb_agg(to_jsonb(changed)) from changed
    ), '[]'::jsonb) else null end as rows from revocation_mfa
  `);
  if (!result || !Array.isArray(result.rows) || result.rows.length > 1) throw new Error("Invalid guarded revocation write");
  if (!result.rows.length) throw notFound();
  const changed = z.object({ id: z.string() }).strict().parse(result.rows[0]);
  if (changed.id !== target.id) throw new Error("Invalid guarded revocation identity");
}

for (const command of revocationCommands) {
  const { path, capability, action } = command;
  tenancyRouter.delete(`/${path}/:id`, async (req, res) => {
    const parsed = id.safeParse(param(req, "id"));
    if (!parsed.success) throw notFound();
    await inTenancy(req, async tx => {
      const current = await lockRevocationTarget(tx, command, parsed.data);
      // New snapshot after any lock wait, including no-match and global rows.
      await requireRevocationOutcomeMfa(tx);
      if (!current?.buildingId) throw notFound();
      await assertCapability(tx, capability, current.buildingId);
      if (!current.active) {
        // We own the row lock already. Repeating the guarded locked read has
        // a fresh snapshot and retains the exact UPDATE-policy visibility,
        // capability and MFA decision for this otherwise empty204 response.
        const unchanged = await lockRevocationTarget(tx, command, current.id);
        if (!unchanged?.buildingId) throw notFound();
        if (!unchanged.allowed) throw forbidden("Sem a capacidade necessária para este recurso");
        return;
      }
      // Audit while the actor still has the grant, including self-revocation.
      // Both statements roll back together if the update is rejected.
      await recordAudit(tx, req, { userId: currentAuth(req).userId, buildingId: current.buildingId, action, resourceType: path, resourceId: current.id });
      await writeRevocation(tx, command, current);
      // Catch changes made by audit/update triggers. Rechecking the capability
      // here would incorrectly roll back a legitimate self-revocation.
      await requireRevocationOutcomeMfa(tx);
    });
    res.status(204).end();
  });
}
