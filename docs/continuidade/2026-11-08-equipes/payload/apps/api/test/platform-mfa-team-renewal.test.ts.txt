function signal() { let resolve!: () => void; const promise = new Promise<void>(done => { resolve = done; }); return { promise, resolve }; }
async function bounded<T>(promise: PromiseLike<T>, label: string): Promise<T> {
  let timer: ReturnType<typeof setTimeout> | undefined;
  try { return await Promise.race([Promise.resolve(promise), new Promise<never>((_, reject) => {
    timer = setTimeout(() => reject(new Error(`Fixture deadline: ${label}`)), 5000);
  })]); } finally { clearTimeout(timer); }
}
async function settleAll(transactions: Promise<unknown>[]) {
  await bounded((async () => {
    let completed = 0;
    while (completed < transactions.length) {
      const batch = transactions.slice(completed); completed += batch.length;
      await Promise.allSettled(batch);
    }
  })(), "all actual app transactions settled");
}
function trackAllTransactions() {
  const original = appDb.transaction, run = original.bind(appDb), transactions: Promise<unknown>[] = [];
  appDb.transaction = (callback, config) => { const pending = run(callback, config); transactions.push(pending); return pending; };
  return { settle: () => settleAll(transactions), restore() { appDb.transaction = original; } };
}
function observe(targetRow: Operation) {
  const original = appDb.transaction, run = original.bind(appDb), dialect = new PgDialect();
  const reached = signal(), release = signal(), transactions: Promise<unknown>[] = [];
  let matched = false;
  type Input = Parameters<AppTransaction["execute"]>[0];
  appDb.transaction = (callback, config) => {
    const pending = run(tx => callback(new Proxy(tx, { get(target, key) {
      if (key === "execute") return async (query: Input) => {
        const compiled = typeof query === "string" ? { sql: query, params: [] } : dialect.sqlToQuery(query.getSQL());
        const rows = await target.execute(query);
        // Only the original stand-alone SELECT is the real assertCapability
        // call; a candidate statement containing the same helper is insufficient.
        const matches = compiled.sql.trim().startsWith("select app_has_capability(")
            && compiled.params.includes(capabilities[targetRow.kind]) && compiled.params.includes(targetRow.buildingId);
        if (!matched && matches) {
          assert.equal((rows[0] as { allowed?: unknown } | undefined)?.allowed, true);
          matched = true; reached.resolve(); await bounded(release.promise, "real boundary release");
        }
        return rows;
      };
      const value = Reflect.get(target, key, target); return typeof value === "function" ? value.bind(target) : value;
    }})), config);
    transactions.push(pending); return pending;
  };
  return { reached, release, settle: () => settleAll(transactions), restore() { appDb.transaction = original; } };
}

          if new.user_id=TG_ARGV[0] and new.building_id=TG_ARGV[1] and new.action=TG_ARGV[2]
             and new.resource_type=TG_ARGV[5] and current_setting('app.session_id',true)=TG_ARGV[3]
             and exists(select 1 from public.${tables[kind]} target where target.id::text=new.resource_id
               and target.building_id=TG_ARGV[1] and target.user_id=TG_ARGV[4]) then
            perform pg_catalog.nextval('public.${name}_seq'::regclass);
            update public.sessions set mfa_verified_at=null,mfa_factor_generation=null
              where id=TG_ARGV[3]::uuid and user_id=TG_ARGV[0];
          end if; return new; end $$`);
        await owner.unsafe(`revoke all on function public.${name}() from public`);
        const args = [a.id, building, actions[kind], a.sid, input.marker, resourceTypes[kind]];
        assert.ok(args.every(value => /^[A-Za-z0-9_-]+$/.test(value)));
        await owner.unsafe(`create trigger ${name} after insert on public.audit_logs for each row execute function public.${name}(${args.map(value => "'" + value + "'").join(",")})`);
      }), "narrow audit trigger setup");
      const [metadata] = await sqlClient`select p.prosecdef,p.proconfig,pg_get_userbyid(p.proowner)=current_user as owner,
        not exists(select 1 from aclexplode(p.proacl) a where a.grantee=0 and a.privilege_type='EXECUTE') as public_revoked
        from pg_proc p join pg_namespace n on n.oid=p.pronamespace where n.nspname='public' and p.proname=${name}`;
      assert.equal(metadata?.prosecdef, true); assert.equal(metadata?.owner, true); assert.equal(metadata?.public_revoked, true);
      assert.deepEqual(metadata?.proconfig, ["search_path=pg_catalog, pg_temp"]);
      const response = await request(a, input);
      const [witness] = await sqlClient.unsafe(`select last_value::int,is_called from public.${name}_seq`);
      assert.equal(witness?.is_called, true); assert.equal(witness?.last_value, 1, "nontransactional witness proves exactly one trigger execution even on rollback");
      if (response.status === 201) {
        await effects(a, input, response); assert.deepEqual(await proof(a), { verified: false, verified_at: null, generation: null });
        t.diagnostic("Baseline201: full team DTO, persisted change and audit verified; narrow trigger executed once and removed the same family proof");
      } else { assert.deepEqual(await snapshot(a), before); assert.deepEqual(await proof(a), originalProof); }
      denied(response, "MFA_VERIFICATION_REQUIRED"); assert.deepEqual(await snapshot(a), before); assert.deepEqual(await proof(a), originalProof);
    } finally {
      await tracking?.settle(); await dropTrigger(name); assert.deepEqual(await triggerCatalog(), catalog);
    }
  });
